Test nâng cao: mock, database thật, fuzz, load test

Test table, fake vs mock (gomock), testcontainers với Postgres thật, golden file, fuzzing, test race, đo tải với k6 / vegeta, coverage trong CI.

Học xong bài này bạn sẽ:
  • Viết test tích hợp với database thật
  • Chọn fake hay mock
  • Đo tải trước khi lên production

#Từ "có test" đến "test đáng tin"

Bài 10 dạy go test, test dạng bảng, httptest. Bài này lo phần khó hơn: test những thứ có phụ thuộc bên ngoài (database, API bên thứ ba, thời gian), test tìm lỗi mà bạn không nghĩ ra (fuzz), và đo xem server chịu được bao nhiêu tải.

Cộng đồng Go nghiêng về fake viết tay. Interface Go nhỏ (bài 6) nên fake thường chỉ vài chục dòng.

service/class.go
package service

import (
    "context"
    "errors"
)

var ErrQuota = errors.New("đã đủ số lớp miễn phí")

type Class struct {
    ID        int64
    TeacherID int64
    Name      string
}

// Interface do service tự khai báo, chỉ gồm những gì service cần
type ClassStore interface {
    CountByTeacher(ctx context.Context, teacherID int64) (int, error)
    Create(ctx context.Context, c Class) (Class, error)
}

type Mailer interface {
    Send(ctx context.Context, to, subject, body string) error
}

type ClassService struct {
    Store     ClassStore
    Mail      Mailer
    FreeQuota int
}

func (s *ClassService) Create(ctx context.Context, teacherID int64, email, name string) (Class, error) {
    n, err := s.Store.CountByTeacher(ctx, teacherID)
    if err != nil {
        return Class{}, err
    }
    if n >= s.FreeQuota {
        return Class{}, ErrQuota
    }
    c, err := s.Store.Create(ctx, Class{TeacherID: teacherID, Name: name})
    if err != nil {
        return Class{}, err
    }
    _ = s.Mail.Send(ctx, email, "Đã tạo lớp", "Lớp "+name+" đã sẵn sàng")
    return c, nil
}

package service

import (
    "context"
    "errors"
    "sync"
    "testing"
)

// Fake store: map trong bộ nhớ, chạy thật
type fakeStore struct {
    mu      sync.Mutex
    classes []Class
    failErr error // để thử trường hợp database lỗi
}

func (f *fakeStore) CountByTeacher(_ context.Context, tid int64) (int, error) {
    f.mu.Lock()
    defer f.mu.Unlock()
    if f.failErr != nil {
        return 0, f.failErr
    }
    n := 0
    for _, c := range f.classes {
        if c.TeacherID == tid {
            n++
        }
    }
    return n, nil
}

func (f *fakeStore) Create(_ context.Context, c Class) (Class, error) {
    f.mu.Lock()
    defer f.mu.Unlock()
    c.ID = int64(len(f.classes) + 1)
    f.classes = append(f.classes, c)
    return c, nil
}

// Mailer ghi lại thư đã gửi: một "mock" viết tay đơn giản
type sentMail struct{ to, subject string }
type recordMailer struct{ sent []sentMail }

func (m *recordMailer) Send(_ context.Context, to, subject, _ string) error {
    m.sent = append(m.sent, sentMail{to, subject})
    return nil
}

func TestCreateClass(t *testing.T) {
    ctx := context.Background()

    t.Run("tạo được lớp đầu tiên và gửi email", func(t *testing.T) {
        st, ml := &fakeStore{}, &recordMailer{}
        svc := &ClassService{Store: st, Mail: ml, FreeQuota: 1}

        c, err := svc.Create(ctx, 7, "co.hoa@example.com", "Lớp 5A")
        if err != nil {
            t.Fatalf("lỗi không mong muốn: %v", err)
        }
        if c.ID == 0 || c.TeacherID != 7 {
            t.Errorf("lớp sai: %+v", c)
        }
        if len(ml.sent) != 1 || ml.sent[0].to != "co.hoa@example.com" {
            t.Errorf("thư đã gửi: %+v", ml.sent)
        }
    })

    t.Run("vượt hạn mức thì báo ErrQuota, không gửi email", func(t *testing.T) {
        st := &fakeStore{classes: []Class{{ID: 1, TeacherID: 7}}}
        ml := &recordMailer{}
        svc := &ClassService{Store: st, Mail: ml, FreeQuota: 1}

        _, err := svc.Create(ctx, 7, "co.hoa@example.com", "Lớp 5B")
        if !errors.Is(err, ErrQuota) {
            t.Fatalf("muốn ErrQuota, nhận %v", err)
        }
        if len(ml.sent) != 0 {
            t.Errorf("không được gửi email, nhưng đã gửi %d", len(ml.sent))
        }
    })

    t.Run("database lỗi thì trả lỗi đó", func(t *testing.T) {
        boom := errors.New("db down")
        svc := &ClassService{Store: &fakeStore{failErr: boom}, Mail: &recordMailer{}, FreeQuota: 1}
        if _, err := svc.Create(ctx, 7, "x@y.z", "A"); !errors.Is(err, boom) {
            t.Fatalf("muốn lỗi db, nhận %v", err)
        }
    })
}

Mock sinh tự động: gomock và mockery

Nhiều công ty dùng mock sinh code, bạn cần đọc được. go.uber.org/mock (gomock):

Go · trích đoạn
//go:generate go run go.uber.org/mock/mockgen -source=class.go -destination=mock_test.go -package=service

func TestCreateWithGomock(t *testing.T) {
    ctrl := gomock.NewController(t)
    st := NewMockClassStore(ctrl)
    ml := NewMockMailer(ctrl)

    // Khai báo trước: phải gọi đúng các hàm này, đúng tham số, đúng số lần
    st.EXPECT().CountByTeacher(gomock.Any(), int64(7)).Return(0, nil)
    st.EXPECT().Create(gomock.Any(), Class{TeacherID: 7, Name: "Lớp 5A"}).Return(Class{ID: 1, TeacherID: 7, Name: "Lớp 5A"}, nil)
    ml.EXPECT().Send(gomock.Any(), "co.hoa@example.com", "Đã tạo lớp", gomock.Any()).Return(nil).Times(1)

    svc := &ClassService{Store: st, Mail: ml, FreeQuota: 1}
    if _, err := svc.Create(context.Background(), 7, "co.hoa@example.com", "Lớp 5A"); err != nil {
        t.Fatal(err)
    }
    // ctrl tự kiểm tra cuối test: thiếu lời gọi nào đã EXPECT thì test lỗi
}

Mock quá chi tiết (khai báo từng lời gọi store) làm test gắn chặt vào cách cài đặt: đổi CountByTeacher thành một câu SQL kiểm hạn mức ngay trong Create là vỡ hết test dù hành vi đúng. Dùng mock cho ranh giới với bên ngoài (email, thanh toán), dùng fake hoặc database thật cho store.

#Test với Postgres thật: testcontainers

Fake store không bắt được lỗi SQL: sai tên cột, thiếu index, vi phạm ràng buộc, khoá dòng. Store phải được test với Postgres thật. testcontainers-go tự bật một container Postgres khi test chạy và tắt khi xong.

package store

import (
    "context"
    "log"
    "os"
    "testing"
    "time"

    "github.com/jackc/pgx/v5/pgxpool"
    "github.com/testcontainers/testcontainers-go"
    "github.com/testcontainers/testcontainers-go/modules/postgres"
    "github.com/testcontainers/testcontainers-go/wait"
)

var testPool *pgxpool.Pool

// TestMain chạy MỘT lần cho cả package: bật Postgres, chạy migration, chạy mọi test, tắt Postgres
func TestMain(m *testing.M) {
    ctx := context.Background()
    pg, err := postgres.Run(ctx, "postgres:17-alpine",
        postgres.WithDatabase("test"),
        postgres.WithUsername("test"),
        postgres.WithPassword("test"),
        postgres.WithInitScripts("../migrations/00001_init.sql"),
        testcontainers.WithWaitStrategy(
            wait.ForLog("database system is ready to accept connections").WithOccurrence(2).WithStartupTimeout(60*time.Second)),
    )
    if err != nil {
        log.Fatal(err)
    }
    dsn, _ := pg.ConnectionString(ctx, "sslmode=disable")
    testPool, err = pgxpool.New(ctx, dsn)
    if err != nil {
        log.Fatal(err)
    }

    code := m.Run()

    testPool.Close()
    _ = pg.Terminate(ctx)
    os.Exit(code)
}

// Mỗi test chạy trong một transaction rồi rollback: dữ liệu test không lẫn sang nhau
func withTx(t *testing.T, fn func(ctx context.Context, q Querier)) {
    t.Helper()
    ctx := context.Background()
    tx, err := testPool.Begin(ctx)
    if err != nil {
        t.Fatal(err)
    }
    t.Cleanup(func() { tx.Rollback(ctx) })
    fn(ctx, tx)
}

#Test API bên thứ ba: httptest.Server

Code gọi API ngân hàng / AI / gửi SMS. Không gọi thật trong test, cũng không cần mock: bật một server HTTP giả ngay trong test và trỏ client vào nó.

payment/client_test.go · trích đoạn
func TestCreatePaymentRetriesOn503(t *testing.T) {
    var calls atomic.Int32
    srv := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
        if r.Header.Get("Idempotency-Key") == "" {
            t.Error("thiếu Idempotency-Key")
        }
        if calls.Add(1) < 3 {
            w.WriteHeader(503) // hai lần đầu lỗi tạm thời
            return
        }
        w.Write([]byte(`{"id":"pay_1","status":"pending"}`))
    }))
    defer srv.Close()

    c := NewClient(srv.URL, "test-key") // client nhận baseURL qua tham số: dễ test
    p, err := c.Create(context.Background(), 49000, "order-1")
    if err != nil {
        t.Fatal(err)
    }
    if p.ID != "pay_1" || calls.Load() != 3 {
        t.Fatalf("p=%+v, calls=%d", p, calls.Load())
    }
}

#Thời gian trong test

Code có "hạn nộp hôm nay", "phiên hết hạn sau 30 ngày", "thử lại sau 1 phút". Đừng gọi time.Now() rải rác: truyền đồng hồ vào.

Go · trích đoạn
type Service struct {
    now func() time.Time // production: time.Now, test: hàm trả giờ cố định
}

func (s *Service) IsLate(due time.Time) bool { return s.now().After(due) }

// test
svc := &Service{now: func() time.Time { return time.Date(2026, 11, 3, 23, 59, 0, 0, vn) }}

Go 1.25 có testing/synctest: chạy test trong một "bong bóng" có đồng hồ giả. time.Sleep(time.Hour) trong bong bóng trả về ngay, nhưng thời gian giả vẫn tiến đúng 1 giờ. Rất hợp để test timeout, ticker, retry có chờ.

retry_test.go · trích đoạn
func TestDebounce(t *testing.T) {
    synctest.Test(t, func(t *testing.T) {
        var d Debouncer
        var n atomic.Int32
        for range 5 {
            d.Call(2*time.Second, func() { n.Add(1) })
            time.Sleep(500 * time.Millisecond) // đồng hồ giả: không chờ thật
        }
        time.Sleep(3 * time.Second)
        synctest.Wait() // chờ mọi goroutine trong bong bóng dừng lại
        if n.Load() != 1 {
            t.Fatalf("debounce phải chỉ chạy 1 lần, chạy %d", n.Load())
        }
    })
}

#Golden file: so sánh với kết quả đã duyệt

Output dài (JSON của báo cáo, file CSV, email HTML): thay vì viết tay chuỗi mong đợi, lưu kết quả đúng vào testdata/*.golden và so sánh. Khi cố ý đổi output, chạy với cờ -update để ghi lại.

Go · trích đoạn
var update = flag.Bool("update", false, "ghi lại file golden")

func TestExportCSV(t *testing.T) {
    got := exportCSV(sampleClass())
    path := filepath.Join("testdata", "export.golden.csv") // go test bỏ qua thư mục testdata khi build
    if *update {
        os.WriteFile(path, got, 0o644)
    }
    want, err := os.ReadFile(path)
    if err != nil {
        t.Fatal(err)
    }
    if !bytes.Equal(got, want) {
        t.Errorf("CSV khác bản đã duyệt. Chạy go test -update rồi xem git diff")
    }
}

Khi review pull request, git diff testdata/ cho thấy chính xác output đổi thế nào. Thư viện so sánh dễ đọc: github.com/google/go-cmp/cmp, cmp.Diff(want, got) in ra từng chỗ khác.

#Fuzz: để máy tìm lỗi

Bạn viết test cho các đầu vào bạn nghĩ ra. Fuzz test cho Go tự sinh hàng triệu đầu vào, ưu tiên những đầu vào đi vào nhánh code mới, để tìm panic hay vi phạm tính chất.

name_test.go
package name

import (
    "strings"
    "testing"
    "unicode/utf8"
)

// Chuẩn hoá tên: bỏ khoảng trắng thừa, viết hoa chữ đầu mỗi từ
func Normalize(s string) string {
    words := strings.Fields(s)
    for i, w := range words {
        words[i] = strings.ToUpper(w[:1]) + strings.ToLower(w[1:]) // lỗi ẩn: cắt theo byte
    }
    return strings.Join(words, " ")
}

func FuzzNormalize(f *testing.F) {
    f.Add("nguyễn văn an") // vài mẫu khởi đầu
    f.Add("  AN  ")
    f.Fuzz(func(t *testing.T, s string) {
        if !utf8.ValidString(s) {
            t.Skip()
        }
        out := Normalize(s)
        // Kiểm tra TÍNH CHẤT, không phải giá trị cụ thể
        if !utf8.ValidString(out) {
            t.Fatalf("kết quả không phải UTF-8 hợp lệ: %q → %q", s, out)
        }
        if Normalize(out) != out {
            t.Fatalf("chuẩn hoá hai lần ra kết quả khác: %q", s)
        }
    })
}

#Bài tập

Fuzz hàm đọc mã lớp

Mã lớp có dạng 8 ký tự gồm chữ in hoa và số, không có O, 0, I, 1 (dễ nhầm). Người dùng có thể gõ chữ thường, có khoảng trắng hoặc dấu gạch: " abcd-5a23 ". Viết ParseCode(s string) (string, error) và fuzz test kiểm tra: không panic, kết quả luôn đúng 8 ký tự hợp lệ khi không lỗi, và ParseCode(kết quả) trả lại chính nó.

Gợi ý từng bước

  1. Duyệt từng rune, bỏ khoảng trắng và -, đổi unicode.ToUpper.
  2. Ký tự hợp lệ: strings.ContainsRune("ABCDEFGHJKLMNPQRSTUVWXYZ23456789", r).
  3. Bẫy: unicode.ToUpper của vài ký tự Unicode lạ ra chữ Latin? Ví dụ chữ ı (i không chấm của tiếng Thổ Nhĩ Kỳ) viết hoa thành I. Fuzz sẽ tìm ra.
  4. Tính chất thứ ba gọi là idempotent: chuẩn hoá hai lần như một lần.
Xem lời giải
code.go
package code

import (
    "errors"
    "strings"
    "unicode"
)

const alphabet = "ABCDEFGHJKLMNPQRSTUVWXYZ23456789"

var ErrBadCode = errors.New("mã lớp gồm 8 chữ hoặc số")

func ParseCode(s string) (string, error) {
    var b strings.Builder
    for _, r := range s {
        if unicode.IsSpace(r) || r == '-' {
            continue
        }
        if r > unicode.MaxASCII { // chặn ký tự ngoài ASCII TRƯỚC khi viết hoa (ı → I)
            return "", ErrBadCode
        }
        r = unicode.ToUpper(r)
        if !strings.ContainsRune(alphabet, r) {
            return "", ErrBadCode
        }
        b.WriteRune(r)
        if b.Len() > 8 {
            return "", ErrBadCode
        }
    }
    if b.Len() != 8 {
        return "", ErrBadCode
    }
    return b.String(), nil
}
code_test.go
package code

import (
    "strings"
    "testing"
)

func FuzzParseCode(f *testing.F) {
    for _, s := range []string{"ABCD5A23", " abcd-5a23 ", "ABCD", "ıııııııı", ""} {
        f.Add(s)
    }
    f.Fuzz(func(t *testing.T, s string) {
        got, err := ParseCode(s)
        if err != nil {
            return // từ chối thì không sao, miễn không panic
        }
        if len(got) != 8 {
            t.Fatalf("%q → %q: không đủ 8 ký tự", s, got)
        }
        for _, r := range got {
            if !strings.ContainsRune(alphabet, r) {
                t.Fatalf("%q → %q: có ký tự %q không hợp lệ", s, got, r)
            }
        }
        again, err := ParseCode(got)
        if err != nil || again != got {
            t.Fatalf("không idempotent: %q → %q → %q, %v", s, got, again, err)
        }
    })
}

func TestParseCode(t *testing.T) {
    cases := map[string]string{" abcd-5a23 ": "ABCD5A23", "ıııııııı": "", "ABCD5A2O": ""}
    for in, want := range cases {
        got, _ := ParseCode(in)
        if got != want {
            t.Errorf("ParseCode(%q) = %q, muốn %q", in, got, want)
        }
    }
}

Test tích hợp: hai request cùng vào lớp gần đầy

Lớp sĩ số 30 đã có 29 em. Hai phụ huynh bấm "vào lớp" cùng lúc. Viết test tích hợp với Postgres thật chứng minh store chỉ cho một người vào. Dùng hàm Join có điều kiện sĩ số nằm trong câu INSERT (bài 19).

Gợi ý từng bước

  1. Không dùng withTx (một transaction) được, vì cần hai kết nối thật chạy song song. Dùng testPool trực tiếp và tự dọn dữ liệu bằng t.Cleanup.
  2. Tạo lớp và 29 em bằng generate_series: INSERT INTO enrollments SELECT $1, g FROM generate_series(1, 29) g.
  3. Hai goroutine cùng gọi Join, chờ nhau bằng một channel start để thật sự cùng lúc.
  4. Đếm: đúng một lỗi ErrFull, đúng 30 em.

store/join.go · trích đoạn
var ErrFull = errors.New("lớp đã đủ sĩ số")

// Điều kiện sĩ số nằm NGAY trong câu INSERT, khoá dòng lớp để hai câu không cùng đếm 29
func (s *ClassStore) Join(ctx context.Context, classID, studentID int64) error {
    tx, err := s.pool.Begin(ctx)
    if err != nil {
        return err
    }
    defer tx.Rollback(ctx)

    var capacity int
    if err := tx.QueryRow(ctx, `SELECT capacity FROM classes WHERE id = $1 FOR UPDATE`, classID).Scan(&capacity); err != nil {
        return err
    }
    tag, err := tx.Exec(ctx, `
        INSERT INTO enrollments (class_id, student_id)
        SELECT $1, $2
        WHERE (SELECT count(*) FROM enrollments WHERE class_id = $1) < $3`, classID, studentID, capacity)
    if err != nil {
        return err
    }
    if tag.RowsAffected() == 0 {
        return ErrFull
    }
    return tx.Commit(ctx)
}

func TestJoinRace(t *testing.T) {
    ctx := context.Background()
    s := &ClassStore{pool: testPool}

    var classID int64
    err := testPool.QueryRow(ctx,
        `INSERT INTO classes (teacher_id, name, capacity) VALUES (1, 'Lớp đua', 30) RETURNING id`).Scan(&classID)
    if err != nil {
        t.Fatal(err)
    }
    t.Cleanup(func() { testPool.Exec(ctx, `DELETE FROM classes WHERE id = $1`, classID) }) // ON DELETE CASCADE dọn enrollments

    if _, err := testPool.Exec(ctx,
        `INSERT INTO enrollments (class_id, student_id) SELECT $1, g FROM generate_series(1, 29) g`, classID); err != nil {
        t.Fatal(err)
    }

    start := make(chan struct{})
    errs := make(chan error, 2)
    for _, sid := range []int64{1001, 1002} {
        go func() {
            <-start // cả hai chờ ở đây rồi cùng chạy
            errs <- s.Join(ctx, classID, sid)
        }()
    }
    close(start)

    var full, ok int
    for range 2 {
        switch err := <-errs; {
        case err == nil:
            ok++
        case errors.Is(err, ErrFull):
            full++
        default:
            t.Fatalf("lỗi lạ: %v", err)
        }
    }
    var n int
    testPool.QueryRow(ctx, `SELECT count(*) FROM enrollments WHERE class_id = $1`, classID).Scan(&n)
    if ok != 1 || full != 1 || n != 30 {
        t.Fatalf("ok=%d full=%d sĩ số=%d, muốn 1, 1, 30", ok, full, n)
    }
}
