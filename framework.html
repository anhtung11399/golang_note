Framework: Chi, Gin, Echo và tài liệu OpenAPI

Khi nào dùng framework, cùng một API viết bằng net/http, Chi và Gin, validate bằng struct tag, sinh tài liệu OpenAPI, CORS.

Học xong bài này bạn sẽ:
  • Đọc hiểu code Gin / Chi trong dự án công ty
  • Validate request bằng thư viện
  • Có tài liệu API cho đội frontend

#Có cần framework không?

Từ Go 1.22, router của thư viện chuẩn (http.ServeMux) đã hiểu method và tham số đường dẫn: mux.HandleFunc("GET /classes/{id}", h), lấy r.PathValue("id"). Với phần lớn API, thư viện chuẩn là đủ, và các bài trước đều viết bằng nó.

Nhưng khi vào công ty, rất nhiều khả năng bạn gặp một trong ba cái tên: Gin (phổ biến nhất), Echo, Chi. Bài này giúp bạn đọc hiểu và viết được cả ba, biết chúng thêm gì so với net/http, và cách sinh tài liệu API cho đội frontend.

So sánh với Node: net/http giống module http của Node nhưng đầy đủ hơn nhiều, Chi giống Express tối giản, Gin / Echo giống Express kèm sẵn body-parser, validator và error handler. Chi được nhiều người ưa vì handler vẫn là kiểu chuẩn: mọi middleware viết cho net/http đều dùng lại được.

#Cùng một API, ba cách viết

API mẫu: GET /classes/{id} trả lớp, POST /classes tạo lớp (tên bắt buộc, 1 đến 60 ký tự, sĩ số 1 đến 80). Có middleware ghi log và nhóm /api/v1.

Thư viện chuẩn

main.go
package main

import (
    "encoding/json"
    "log/slog"
    "net/http"
    "strconv"
    "time"
)

type Class struct {
    ID       int64  `json:"id"`
    Name     string `json:"name"`
    Capacity int    `json:"capacity"`
}

func logger(next http.Handler) http.Handler {
    return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
        start := time.Now()
        next.ServeHTTP(w, r)
        slog.Info("req", "method", r.Method, "path", r.URL.Path, "dur", time.Since(start))
    })
}

func writeJSON(w http.ResponseWriter, code int, v any) {
    w.Header().Set("Content-Type", "application/json")
    w.WriteHeader(code)
    json.NewEncoder(w).Encode(v)
}

func main() {
    api := http.NewServeMux()
    api.HandleFunc("GET /classes/{id}", func(w http.ResponseWriter, r *http.Request) {
        id, err := strconv.ParseInt(r.PathValue("id"), 10, 64)
        if err != nil {
            writeJSON(w, 400, map[string]string{"error": "id không hợp lệ"})
            return
        }
        writeJSON(w, 200, Class{ID: id, Name: "Lớp 5A", Capacity: 30})
    })
    api.HandleFunc("POST /classes", func(w http.ResponseWriter, r *http.Request) {
        var in Class
        if err := json.NewDecoder(r.Body).Decode(&in); err != nil {
            writeJSON(w, 400, map[string]string{"error": "JSON không hợp lệ"})
            return
        }
        if n := len([]rune(in.Name)); n < 1 || n > 60 || in.Capacity < 1 || in.Capacity > 80 {
            writeJSON(w, 422, map[string]string{"error": "tên 1-60 ký tự, sĩ số 1-80"})
            return
        }
        in.ID = 1
        writeJSON(w, 201, in)
    })

    root := http.NewServeMux()
    root.Handle("/api/v1/", http.StripPrefix("/api/v1", api)) // nhóm route bằng tiền tố
    http.ListenAndServe(":8080", logger(root))
}

package main

import (
    "encoding/json"
    "net/http"
    "strconv"

    "github.com/go-chi/chi/v5"
    "github.com/go-chi/chi/v5/middleware"
)

// ... type Class và writeJSON như trên

func main() {
    r := chi.NewRouter()
    r.Use(middleware.RequestID) // gắn X-Request-Id
    r.Use(middleware.RealIP)    // lấy IP thật từ X-Forwarded-For (chỉ khi đứng sau proxy tin cậy)
    r.Use(middleware.Logger)
    r.Use(middleware.Recoverer) // panic → 500 thay vì sập

    r.Route("/api/v1", func(r chi.Router) {
        r.Route("/classes", func(r chi.Router) {
            r.Post("/", createClass)
            r.Get("/{id}", func(w http.ResponseWriter, r *http.Request) {
                id, err := strconv.ParseInt(chi.URLParam(r, "id"), 10, 64) // hoặc r.PathValue("id")
                if err != nil {
                    writeJSON(w, 400, map[string]string{"error": "id không hợp lệ"})
                    return
                }
                writeJSON(w, 200, Class{ID: id, Name: "Lớp 5A", Capacity: 30})
            })
        })
    })
    http.ListenAndServe(":8080", r)
}

func createClass(w http.ResponseWriter, r *http.Request) {
    var in Class
    if err := json.NewDecoder(r.Body).Decode(&in); err != nil {
        writeJSON(w, 400, map[string]string{"error": "JSON không hợp lệ"})
        return
    }
    // validate như bản thư viện chuẩn, hoặc dùng go-playground/validator (phần dưới)
    in.ID = 1
    writeJSON(w, 201, in)
}

Chi gần như là thư viện chuẩn có thêm cách nhóm route đẹp hơn và bộ middleware có sẵn. Handler vẫn là http.HandlerFunc, nên code ở các bài trước chép sang dùng ngay.

Gin

Terminal
go get github.com/gin-gonic/gin
main.go
package main

import (
    "net/http"
    "strconv"

    "github.com/gin-gonic/gin"
)

type Class struct {
    ID       int64  `json:"id"`
    Name     string `json:"name" binding:"required,min=1,max=60"`
    Capacity int    `json:"capacity" binding:"required,gte=1,lte=80"`
}

func main() {
    r := gin.New()
    r.Use(gin.Logger(), gin.Recovery())

    v1 := r.Group("/api/v1")
    {
        v1.GET("/classes/:id", func(c *gin.Context) {
            id, err := strconv.ParseInt(c.Param("id"), 10, 64)
            if err != nil {
                c.JSON(http.StatusBadRequest, gin.H{"error": "id không hợp lệ"})
                return
            }
            c.JSON(http.StatusOK, Class{ID: id, Name: "Lớp 5A", Capacity: 30})
        })
        v1.POST("/classes", func(c *gin.Context) {
            var in Class
            if err := c.ShouldBindJSON(&in); err != nil { // đọc JSON + kiểm tra tag binding
                c.JSON(http.StatusUnprocessableEntity, gin.H{"error": err.Error()})
                return
            }
            in.ID = 1
            c.JSON(http.StatusCreated, in)
        })
    }
    r.Run(":8080")
}

Giải thích: thông báo lỗi mặc định của validator là tiếng Anh, khó đọc với người dùng. Phần "Validate bằng struct tag" bên dưới chỉ cách đổi thành câu tiếng Việt theo từng trường.

Những điều riêng của Gin cần biết khi đọc code:

  • gin.H chỉ là map[string]any.
  • c.JSON(...) không dừng handler. Phải return ngay sau đó, giống http.Error.
  • Trong middleware, c.Next() gọi phần tiếp theo, c.Abort() / c.AbortWithStatusJSON(401, ...) chặn không cho chạy tiếp.
  • c.Set("userID", id) / c.GetInt64("userID") thay cho context.WithValue. Nhưng khi gọi xuống service / store, vẫn truyền c.Request.Context(), không truyền c.
  • ShouldBindJSON trả lỗi để bạn tự xử lý. BindJSON (không có Should) tự ghi 400 và bạn không đổi được nội dung lỗi, nên hay dùng ShouldBind....
  • gin.Context được dùng lại cho request khác sau khi handler xong. Nếu mở goroutine bên trong handler, dùng c.Copy().

Echo

main.go
package main

import (
    "net/http"
    "strconv"

    "github.com/labstack/echo/v4"
    "github.com/labstack/echo/v4/middleware"
)

type Class struct {
    ID       int64  `json:"id"`
    Name     string `json:"name"`
    Capacity int    `json:"capacity"`
}

func main() {
    e := echo.New()
    e.Use(middleware.Logger(), middleware.Recover())

    v1 := e.Group("/api/v1")
    v1.GET("/classes/:id", func(c echo.Context) error {
        id, err := strconv.ParseInt(c.Param("id"), 10, 64)
        if err != nil {
            return echo.NewHTTPError(http.StatusBadRequest, "id không hợp lệ") // trả lỗi, Echo tự ghi
        }
        return c.JSON(http.StatusOK, Class{ID: id, Name: "Lớp 5A", Capacity: 30})
    })
    e.Logger.Fatal(e.Start(":8080"))
}

Điểm hay của Echo: handler trả về error, mọi lỗi đi qua một e.HTTPErrorHandler chung. Đó đúng là mẫu "một định dạng lỗi cho toàn bộ API" ở bài 13, có sẵn.

#Gin dùng với kiến trúc nhiều lớp

Framework chỉ nên nằm ở lớp handler. Service và store (bài 14) không được import gin. Nhờ vậy đổi framework hay viết test cho service không cần đụng tới HTTP.

Go · trích đoạn
type ClassHandler struct{ svc *service.ClassService }

func (h *ClassHandler) Create(c *gin.Context) {
    var in createClassReq
    if err := c.ShouldBindJSON(&in); err != nil {
        writeValidation(c, err)
        return
    }
    userID := c.GetInt64("userID") // middleware xác thực đã gắn
    cls, err := h.svc.Create(c.Request.Context(), userID, in.Name, in.Capacity)
    if err != nil {
        writeErr(c, err) // ánh xạ lỗi nghiệp vụ → status code, như bài trước
        return
    }
    c.JSON(http.StatusCreated, cls)
}

func Auth(sessions *auth.Store) gin.HandlerFunc {
    return func(c *gin.Context) {
        uid, err := sessions.UserID(c.Request.Context(), c.GetHeader("Authorization"))
        if err != nil {
            c.AbortWithStatusJSON(401, gin.H{"error": gin.H{"code": "UNAUTHORIZED", "message": "Bạn cần đăng nhập"}})
            return
        }
        c.Set("userID", uid)
        c.Next()
    }
}

// main:
r := gin.New()
r.Use(gin.Recovery())
api := r.Group("/api/v1", Auth(sessions))
api.POST("/classes", classH.Create)

#Validate bằng struct tag

Gin dùng sẵn thư viện github.com/go-playground/validator/v10. Bạn cũng dùng được nó ở thư viện chuẩn hoặc Chi. Các tag hay dùng:

TagÝ nghĩa
requiredkhông được là giá trị rỗng (chuỗi rỗng, 0, nil)
min=1,max=60chuỗi: số ký tự, số: giá trị, slice: số phần tử
gte=1,lte=80lớn hơn hoặc bằng, nhỏ hơn hoặc bằng
email, url, uuidđúng định dạng
oneof=draft publishedchỉ nhận một trong các giá trị
datetime=2006-01-02ngày theo định dạng Go
dive,requiredkiểm tra từng phần tử của slice
omitempty,emailkhông bắt buộc, nhưng có thì phải đúng email

Bẫy với required trên số: capacity: 0 bị coi là "không có". Nếu 0 là giá trị hợp lệ (ví dụ điểm 0), dùng con trỏ *int với required, hoặc bỏ required và dùng gte=0.

Đổi lỗi tiếng Anh thành câu tiếng Việt theo từng trường, trả về định dạng mà frontend dễ gắn vào form:

Go · trích đoạn
import "github.com/go-playground/validator/v10"

var msgs = map[string]string{
    "required": "Không được để trống",
    "min":      "Quá ngắn",
    "max":      "Quá dài",
    "gte":      "Quá nhỏ",
    "lte":      "Quá lớn",
    "email":    "Email không đúng định dạng",
}

func writeValidation(c *gin.Context, err error) {
    var ve validator.ValidationErrors
    if !errors.As(err, &ve) {
        c.JSON(400, gin.H{"error": gin.H{"code": "BAD_JSON", "message": "Dữ liệu gửi lên không đúng"}})
        return
    }
    fields := map[string]string{}
    for _, fe := range ve {
        m := msgs[fe.Tag()]
        if m == "" {
            m = "Không hợp lệ"
        }
        fields[jsonName(fe)] = m
    }
    c.JSON(422, gin.H{"error": gin.H{"code": "VALIDATION", "message": "Vui lòng kiểm tra lại", "fields": fields}})
}

jsonName lấy tên trong tag json thay vì tên trường Go. Cách gọn nhất là đăng ký một lần lúc khởi động: v.RegisterTagNameFunc(func(f reflect.StructField) string { return strings.Split(f.Tag.Get("json"), ",")[0] }), sau đó fe.Field() trả luôn "capacity" thay vì "Capacity".

#CORS

Frontend chạy ở http://localhost:5173, API ở :8080: khác origin, trình duyệt chặn nếu API không trả header CORS. Trình duyệt gửi trước một request OPTIONS (preflight) để hỏi.

Go · trích đoạn
// Chi: github.com/go-chi/cors
r.Use(cors.Handler(cors.Options{
    AllowedOrigins:   []string{"https://app.sobaitap.vn", "http://localhost:5173"},
    AllowedMethods:   []string{"GET", "POST", "PATCH", "DELETE"},
    AllowedHeaders:   []string{"Content-Type", "Authorization", "Idempotency-Key"},
    AllowCredentials: true, // gửi cookie: KHÔNG được dùng AllowedOrigins "*"
    MaxAge:           600,  // trình duyệt nhớ kết quả preflight 10 phút
}))

// Gin: github.com/gin-contrib/cors có cấu hình tương tự

Cách tránh hẳn CORS: cho frontend và API cùng một origin. Khi phát triển, dùng proxy của Vite / Angular (/api → :8080). Khi chạy thật, đặt cả hai sau một tên miền (app.com phục vụ file tĩnh, app.com/api tới Go). Không có CORS thì cũng bớt một chỗ cấu hình sai.

#Tài liệu API với OpenAPI

OpenAPI (tên cũ Swagger) là một file YAML / JSON mô tả mọi endpoint: đường dẫn, tham số, body, các response. Từ file đó có thể: hiện trang tài liệu bấm thử được (Swagger UI, Redoc, Scalar), sinh kiểu TypeScript cho frontend, sinh code server, và kiểm tra request trong test.

Có hai trường phái:

Code trước (code-first)Spec trước (spec-first)
Cách làmviết handler, thêm chú thích, công cụ đọc chú thích sinh ra specviết file openapi.yaml trước, sinh code Go từ nó
Công cụswaggo/swag (Gin, Echo, Chi), humaoapi-codegen, ogen
Ưunhanh, quenspec luôn đúng với code (code sinh ra từ spec), frontend làm song song được
Nhượcchú thích dễ lệch với code thật mà không ai biếtphải học viết YAML

Code sinh ra có kiểu api.Class và một interface. Bạn chỉ việc viết hàm thoả interface đó. Thiếu endpoint nào, trình biên dịch báo lỗi ngay:

type Server struct{ svc *service.ClassService }

// Kiểm tra lúc biên dịch: Server phải có đủ mọi method của interface sinh ra
var _ api.StrictServerInterface = (*Server)(nil)

func (s *Server) GetClass(ctx context.Context, req api.GetClassRequestObject) (api.GetClassResponseObject, error) {
    c, err := s.svc.Get(ctx, req.Id)
    if errors.Is(err, service.ErrNotFound) {
        return api.GetClass404JSONResponse{Error: struct{ Code, Message string }{"NOT_FOUND", "Không tìm thấy lớp"}}, nil
    }
    if err != nil {
        return nil, err
    }
    return api.GetClass200JSONResponse{Id: c.ID, Name: c.Name, Capacity: c.Capacity}, nil
}

// main:
h := api.HandlerFromMux(api.NewStrictHandler(&Server{svc}, nil), http.NewServeMux())

Kiểu struct lồng nhau trong ví dụ trên do code sinh tạo ra. Trong dự án thật, đặt Error thành một schema riêng có tên thì code sinh gọn hơn nhiều.

#Chọn gì?

  • Dự án mới, đội nhỏ: thư viện chuẩn hoặc Chi. Ít phụ thuộc, code của bạn dùng được mãi.
  • Công ty đã dùng Gin / Echo: theo công ty. Giữ framework ở lớp handler.
  • Có đội frontend riêng, API nhiều: spec-first với oapi-codegen, frontend sinh kiểu từ cùng file.
  • Tránh: framework "làm tất cả" kiểu Rails (Beego, Buffalo) cho API mới. Cộng đồng Go thích ghép các thư viện nhỏ.

Câu hỏi nhanh. Trong handler Gin, gọi c.JSON(400, ...) khi lỗi validate rồi quên return. Chuyện gì xảy ra?

c.JSON chỉ ghi response. Muốn dừng phải return. Trong middleware thì dùng c.Abort... để các handler sau không chạy.

Câu hỏi nhanh. API trả Access-Control-Allow-Origin: * và frontend gửi kèm cookie (credentials: 'include'). Kết quả?

Đây là quy định của trình duyệt. Server phải đối chiếu Origin với danh sách cho phép và trả lại đúng origin đó, kèm Access-Control-Allow-Credentials: true.

Câu hỏi nhanh. Ưu điểm lớn nhất của spec-first (oapi-codegen strict server) so với chú thích swag?

Với swag, chú thích chỉ là ghi chú, đổi code mà quên sửa chú thích thì tài liệu sai mà không ai biết.

#Bài tập

Viết lại API lớp học bằng Gin, có lỗi tiếng Việt theo trường

Viết API Gin có POST /api/v1/classes và GET /api/v1/classes/:id, lưu trong bộ nhớ (map có mutex). Yêu cầu:

  • Tên bắt buộc 1 đến 60 ký tự, sĩ số 1 đến 80, năm học dạng 2026-2027.
  • Lỗi validate trả 422 với fields theo tên JSON, câu tiếng Việt.
  • Không tìm thấy trả 404 theo định dạng {error:{code,message}}.

Gợi ý từng bước

  1. Năm học: tag binding:"required,len=9" rồi tự kiểm tra thêm năm sau = năm trước + 1. Hoặc đăng ký validator riêng.
  2. Đăng ký tên trường theo tag json: lấy validator của Gin bằng binding.Validator.Engine().(*validator.Validate).
  3. Map lưu lớp: map[int64]Class + sync.Mutex + bộ đếm id. Gin chạy mỗi request ở một goroutine nên bắt buộc có mutex.
  4. Đăng ký validator riêng: v.RegisterValidation("schoolyear", func(fl validator.FieldLevel) bool {...}), rồi dùng tag binding:"required,schoolyear".

main.go
package main

import (
    "errors"
    "net/http"
    "reflect"
    "strconv"
    "strings"
    "sync"

    "github.com/gin-gonic/gin"
    "github.com/gin-gonic/gin/binding"
    "github.com/go-playground/validator/v10"
)

type Class struct {
    ID         int64  `json:"id"`
    Name       string `json:"name" binding:"required,min=1,max=60"`
    Capacity   int    `json:"capacity" binding:"required,gte=1,lte=80"`
    SchoolYear string `json:"schoolYear" binding:"required,schoolyear"`
}

var msgs = map[string]string{
    "required":   "Không được để trống",
    "min":        "Quá ngắn",
    "max":        "Quá dài (tối đa {p} ký tự)",
    "gte":        "Tối thiểu {p}",
    "lte":        "Tối đa {p}",
    "schoolyear": "Năm học dạng 2026-2027",
}

func schoolYear(fl validator.FieldLevel) bool {
    a, b, ok := strings.Cut(fl.Field().String(), "-")
    y1, e1 := strconv.Atoi(a)
    y2, e2 := strconv.Atoi(b)
    return ok && e1 == nil && e2 == nil && len(a) == 4 && y2 == y1+1
}

type store struct {
    mu     sync.Mutex
    nextID int64
    m      map[int64]Class
}

func apiErr(c *gin.Context, status int, code, msg string) {
    c.JSON(status, gin.H{"error": gin.H{"code": code, "message": msg}})
}

func main() {
    if v, ok := binding.Validator.Engine().(*validator.Validate); ok {
        v.RegisterTagNameFunc(func(f reflect.StructField) string {
            return strings.Split(f.Tag.Get("json"), ",")[0]
        })
        v.RegisterValidation("schoolyear", schoolYear)
    }

    s := &store{m: map[int64]Class{}}
    r := gin.New()
    r.Use(gin.Recovery())
    v1 := r.Group("/api/v1")

    v1.POST("/classes", func(c *gin.Context) {
        var in Class
        if err := c.ShouldBindJSON(&in); err != nil {
            var ve validator.ValidationErrors
            if !errors.As(err, &ve) {
                apiErr(c, 400, "BAD_JSON", "Dữ liệu gửi lên không đúng")
                return
            }
            fields := gin.H{}
            for _, fe := range ve {
                m := msgs[fe.Tag()]
                if m == "" {
                    m = "Không hợp lệ"
                }
                fields[fe.Field()] = strings.ReplaceAll(m, "{p}", fe.Param())
            }
            c.JSON(422, gin.H{"error": gin.H{"code": "VALIDATION", "message": "Vui lòng kiểm tra lại", "fields": fields}})
            return
        }
        s.mu.Lock()
        s.nextID++
        in.ID = s.nextID
        s.m[in.ID] = in
        s.mu.Unlock()
        c.JSON(http.StatusCreated, in)
    })

    v1.GET("/classes/:id", func(c *gin.Context) {
        id, err := strconv.ParseInt(c.Param("id"), 10, 64)
        if err != nil {
            apiErr(c, 400, "BAD_ID", "Mã lớp không hợp lệ")
            return
        }
        s.mu.Lock()
        cls, ok := s.m[id]
        s.mu.Unlock()
        if !ok {
            apiErr(c, 404, "NOT_FOUND", "Không tìm thấy lớp")
            return
        }
        c.JSON(200, cls)
    })

    r.Run(":8080")
}

Viết openapi.yaml cho API nộp bài và sinh kiểu TypeScript

Viết spec cho GET /assignments?classId=&page= (trả danh sách có phân trang) và POST /assignments/{id}/submit (body: danh sách khoá ảnh, trả 201 hoặc 409 nếu đã chấm). Sinh kiểu TypeScript và viết một lời gọi.

Gợi ý từng bước

  1. Tham số query: in: query. classId bắt buộc, page không bắt buộc, mặc định 1.
  2. Response phân trang: object có items (mảng Assignment), page, totalPages.
  3. Body: requestBody với schema có fileKeys: mảng string, minItems: 1, maxItems: 5.
  4. Dùng lại schema Error cho 409 bằng $ref.
