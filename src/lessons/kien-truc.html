Middleware: những lớp vỏ quanh handler

Có những việc mọi request đều cần: ghi log, gắn request id, chặn panic, kiểm tra đăng nhập, giới hạn tần suất. Thay vì chép vào từng handler, ta bọc handler trong các middleware.

Trong Go, middleware chỉ là một hàm nhận http.Handler và trả về http.Handler mới:

type Middleware func(http.Handler) http.Handler
[ext] RequestID :: gắn mã cho request, đưa vào ctx và header [ext] Logger :: đo thời gian, ghi status khi response xong [warn] Recover :: chặn panic, trả 500 tử tế [q] Auth :: đọc cookie phiên, đưa người dùng vào ctx [go] Handler :: nghiệp vụ của endpoint

Giống hệt interceptor của Angular HttpClient hay middleware của Express / Next.js, chỉ là ở phía server. Nếu bạn từng viết app.use((req, res, next) => ...) trong Express, thì next.ServeHTTP(w, r) chính là next().

Request ID

func RequestID(next http.Handler) http.Handler {
    return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
        id := r.Header.Get("X-Request-ID")
        if id == "" || len(id) > 64 {
            id = newID() // ví dụ 16 byte ngẫu nhiên dạng hex
        }
        w.Header().Set("X-Request-ID", id)
        ctx := WithRequestID(r.Context(), id) // hàm ở bài 10
        next.ServeHTTP(w, r.WithContext(ctx))
    })
}

Khi người dùng báo lỗi kèm mã request (hiện trên màn hình lỗi), bạn tìm đúng dòng log của request đó trong hàng triệu dòng.

Logger: cần biết status code

http.ResponseWriter không cho đọc lại status đã ghi. Mẹo chuẩn: bọc nó trong một struct ghi nhớ status.

type statusRecorder struct {
    http.ResponseWriter // nhúng: mọi method khác giữ nguyên
    status int
    bytes  int
}

func (s *statusRecorder) WriteHeader(code int) {
    s.status = code
    s.ResponseWriter.WriteHeader(code)
}

func (s *statusRecorder) Write(b []byte) (int, error) {
    if s.status == 0 {
        s.status = http.StatusOK
    }
    n, err := s.ResponseWriter.Write(b)
    s.bytes += n
    return n, err
}

func Logger(log *slog.Logger) Middleware {
    return func(next http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
            start := time.Now()
            rec := &statusRecorder{ResponseWriter: w}
            next.ServeHTTP(rec, r)
            log.InfoContext(r.Context(), "http",
                "method", r.Method,
                "path", r.URL.Path,
                "status", rec.status,
                "bytes", rec.bytes,
                "duration_ms", time.Since(start).Milliseconds(),
                "request_id", RequestIDFrom(r.Context()),
            )
        })
    }
}

Ghép các middleware

func Chain(h http.Handler, mws ...Middleware) http.Handler {
    for i := len(mws) - 1; i >= 0; i-- { // bọc từ trong ra ngoài
        h = mws[i](h)
    }
    return h
}

handler := Chain(mux, RequestID, Logger(logger), Recover)
// thứ tự chạy khi có request: RequestID → Logger → Recover → mux

Chạy thử để thấy thứ tự. Chương trình dưới không mở cổng mạng: nó dùng httptest (bài 10) để gửi request giả thẳng vào handler, nên chạy được cả trên Go Playground.

package main

import (
    "fmt"
    "net/http"
    "net/http/httptest"
)

type Middleware func(http.Handler) http.Handler

// Mỗi middleware in một dòng khi request đi vào, và một dòng khi đi ra
func Outer(next http.Handler) http.Handler {
    return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
        fmt.Println("-> Outer: vào")
        next.ServeHTTP(w, r) // chuyển cho lớp bên trong, giống next() của Express
        fmt.Println("<- Outer: ra")
    })
}

func Auth(next http.Handler) http.Handler {
    return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
        fmt.Println("  -> Auth: kiểm tra đăng nhập")
        if r.Header.Get("Cookie") == "" {
            fmt.Println("  <- Auth: chặn, trả 401, KHÔNG gọi next")
            w.WriteHeader(http.StatusUnauthorized)
            return
        }
        next.ServeHTTP(w, r)
        fmt.Println("  <- Auth: ra")
    })
}

func Chain(h http.Handler, mws ...Middleware) http.Handler {
    for i := len(mws) - 1; i >= 0; i-- { // bọc từ trong ra ngoài
        h = mws[i](h)
    }
    return h
}

func main() {
    hello := http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
        fmt.Println("    handler: xử lý nghiệp vụ")
        w.Write([]byte("xin chào"))
    })
    app := Chain(hello, Outer, Auth) // Outer bọc ngoài cùng

    fmt.Println("== Request có cookie")
    req := httptest.NewRequest("GET", "/", nil)
    req.Header.Set("Cookie", "session=abc")
    rec := httptest.NewRecorder()
    app.ServeHTTP(rec, req)
    fmt.Println("status:", rec.Code, "body:", rec.Body.String())

    fmt.Println("== Request không có cookie")
    rec = httptest.NewRecorder()
    app.ServeHTTP(rec, httptest.NewRequest("GET", "/", nil))
    fmt.Println("status:", rec.Code)
}

== Request có cookie
-> Outer: vào
  -> Auth: kiểm tra đăng nhập
    handler: xử lý nghiệp vụ
  <- Auth: ra
<- Outer: ra
status: 200 body: xin chào
== Request không có cookie
-> Outer: vào
  -> Auth: kiểm tra đăng nhập
  <- Auth: chặn, trả 401, KHÔNG gọi next
<- Outer: ra
status: 401

Vào theo thứ tự Outer → Auth → handler, ra theo thứ tự ngược lại, đúng như hình ở đầu bài. Middleware không gọi next.ServeHTTP là chặn request lại: handler không bao giờ chạy.

Middleware chỉ áp dụng cho một số route (ví dụ yêu cầu đăng nhập) thì bọc riêng route đó: mux.Handle("GET /v1/me", RequireAuth(http.HandlerFunc(me))). Thư viện chi có cú pháp nhóm route tiện hơn (r.Group, r.With), một lý do khiến nó vẫn được dùng nhiều.

Log có cấu trúc với slog

Log dạng chữ tự do (log.Printf("user %d tạo lớp %s", ...)) đọc bằng mắt thì được, nhưng không lọc được khi có hàng triệu dòng. Log có cấu trúc ghi từng trường riêng, thường dạng JSON, để hệ thống log (Grafana Loki, Elasticsearch, Datadog, Cloud Logging...) tìm kiếm và đếm.

logger := slog.New(slog.NewJSONHandler(os.Stdout, &slog.HandlerOptions{Level: slog.LevelInfo}))
slog.SetDefault(logger)

slog.Info("tạo lớp", "class_id", 42, "teacher_id", 7)
slog.Error("gửi email thất bại", "err", err, "user_id", 7)

{"time":"2026-11-03T08:30:00.123Z","level":"INFO","msg":"tạo lớp","class_id":42,"teacher_id":7}

Không bao giờ ghi vào log: mật khẩu, token phiên, mã OTP, số thẻ, nội dung body đầy đủ của request đăng nhập / thanh toán. Log thường được nhiều người đọc hơn database và lưu ở nhiều nơi hơn. Rò rỉ qua log là một kiểu sự cố bảo mật rất phổ biến.

Tách lớp: handler, service, store

Khi handler vừa đọc JSON, vừa kiểm quyền, vừa viết SQL, vừa gửi email, nó dài 200 dòng và không test được. Cách tổ chức phổ biến nhất ở backend Go:

LớpLàm gìKhông làm gì
Handler (httpapi)Đọc request, gọi service, đổi kết quả / lỗi thành responseKhông viết SQL, không chứa quy tắc nghiệp vụ
ServiceQuy tắc nghiệp vụ: quyền, giới hạn, tính toán, điều phối nhiều storeKhông biết HTTP (không có http.Request), không viết SQL
Store (repository)Đọc / ghi database, RedisKhông quyết định nghiệp vụ

Ví dụ (trích đoạn từ một dự án, bài tập cuối bài có bản chạy được đầy đủ): giáo viên gói miễn phí chỉ được tạo 1 lớp.

type ClassStore interface {
    CountByTeacher(ctx context.Context, teacherID int64) (int, error)
    Create(ctx context.Context, c domain.Class) (domain.Class, error)
}

type Classes struct {
    store ClassStore
    clock func() time.Time // tiêm được thời gian → test được các trường hợp theo ngày
}

func NewClasses(s ClassStore) *Classes {
    return &Classes{store: s, clock: time.Now}
}

const FreeClassQuota = 1

func (s *Classes) Create(ctx context.Context, teacherID int64, in CreateClassInput) (domain.Class, error) {
    if err := in.Validate(); err != nil {
        return domain.Class{}, err
    }
    n, err := s.store.CountByTeacher(ctx, teacherID)
    if err != nil {
        return domain.Class{}, fmt.Errorf("đếm lớp: %w", err)
    }
    if n >= FreeClassQuota {
        return domain.Class{}, domain.ErrQuotaExceeded
    }
    return s.store.Create(ctx, domain.Class{TeacherID: teacherID, Name: in.Name, CreatedAt: s.clock()})
}

func (h *Handler) createClass(w http.ResponseWriter, r *http.Request) {
    user := MustUser(r.Context()) // middleware Auth đã đưa vào ctx
    var in service.CreateClassInput
    if err := DecodeJSON(w, r, &in); err != nil {
        WriteError(w, r, err)
        return
    }
    c, err := h.classes.Create(r.Context(), user.ID, in)
    if err != nil {
        WriteError(w, r, err) // ErrQuotaExceeded → 402 hoặc 409 tuỳ quy ước, ở MỘT chỗ
        return
    }
    WriteJSON(w, http.StatusCreated, toClassDTO(c))
}

Đoạn service trên có lỗi tranh chấp: hai request "tạo lớp" tới cùng lúc, cả hai đều đếm được 0 rồi cùng tạo, giáo viên có 2 lớp miễn phí. Bài 16 sửa nó bằng cách đưa điều kiện vào chính câu SQL. Bạn sẽ gặp kiểu lỗi "kiểm tra rồi mới làm" này ở mọi nơi.

Đừng tách lớp cho có. Endpoint chỉ đọc một bảng và trả về thì handler gọi thẳng store là đủ. Cấu trúc ba lớp đáng giá khi có quy tắc nghiệp vụ thật.

Cấu hình qua biến môi trường

Theo nguyên tắc 12-factor app, cấu hình (địa chỉ database, khoá bí mật, cổng) đến từ biến môi trường, không nằm trong code. Cùng một file chạy, đổi biến môi trường là chạy được ở máy bạn, staging hay production.

type Config struct {
    Addr        string
    DatabaseURL string
    RedisURL    string
    SessionKey  []byte
    Env         string // "dev" | "prod"
}

func Load() (Config, error) {
    c := Config{
        Addr:        getenv("ADDR", ":8080"),
        DatabaseURL: os.Getenv("DATABASE_URL"),
        RedisURL:    os.Getenv("REDIS_URL"),
        Env:         getenv("APP_ENV", "dev"),
    }
    var missing []string
    if c.DatabaseURL == "" {
        missing = append(missing, "DATABASE_URL")
    }
    key := os.Getenv("SESSION_KEY")
    if len(key) < 32 {
        missing = append(missing, "SESSION_KEY (ít nhất 32 ký tự)")
    }
    c.SessionKey = []byte(key)
    if len(missing) > 0 {
        return c, fmt.Errorf("thiếu cấu hình: %s", strings.Join(missing, ", "))
    }
    return c, nil
}

func getenv(k, def string) string {
    if v := os.Getenv(k); v != "" {
        return v
    }
    return def
}

Hỏng sớm. Thiếu cấu hình thì dừng ngay lúc khởi động với thông báo rõ ràng, thay vì chạy được rồi lỗi ở request đầu tiên lúc 2 giờ sáng. Bí mật (khoá, mật khẩu database) không bao giờ commit vào git: dùng tính năng secrets của nơi triển khai.

Ghép mọi thứ trong main

Hàm main là nơi duy nhất biết mọi thành phần cụ thể: đọc cấu hình, tạo kết nối, "tiêm" chúng vào service và handler qua các hàm New.... Không cần framework dependency injection.

Trích đoạn. Phần pgxpool (database) học ở bài 15. Bản chạy được không cần database nằm trong lời giải bài tập cuối bài.

func main() {
    if err := run(); err != nil {
        slog.Error("dừng", "err", err)
        os.Exit(1)
    }
}

func run() error {
    cfg, err := config.Load()
    if err != nil {
        return err
    }
    logger := slog.New(slog.NewJSONHandler(os.Stdout, nil))

    // Huỷ ctx khi nhận Ctrl+C (SIGINT) hoặc lệnh dừng của Docker / Kubernetes (SIGTERM)
    ctx, stop := signal.NotifyContext(context.Background(), os.Interrupt, syscall.SIGTERM)
    defer stop()

    db, err := pgxpool.New(ctx, cfg.DatabaseURL)
    if err != nil {
        return fmt.Errorf("kết nối database: %w", err)
    }
    defer db.Close()

    store := postgres.NewStore(db)
    classes := service.NewClasses(store)
    h := httpapi.New(classes, logger)

    srv := &http.Server{
        Addr:              cfg.Addr,
        Handler:           h.Routes(),
        ReadHeaderTimeout: 5 * time.Second,
        ReadTimeout:       15 * time.Second,
        WriteTimeout:      30 * time.Second,
        IdleTimeout:       60 * time.Second,
    }

    errCh := make(chan error, 1)
    go func() {
        logger.Info("bắt đầu", "addr", cfg.Addr)
        errCh <- srv.ListenAndServe()
    }()

    select {
    case err := <-errCh: // server không mở được cổng, v.v.
        return err
    case <-ctx.Done(): // nhận tín hiệu dừng
    }

    logger.Info("đang tắt, chờ các request dở dang")
    shutdownCtx, cancel := context.WithTimeout(context.Background(), 20*time.Second)
    defer cancel()
    return srv.Shutdown(shutdownCtx)
}

Tắt server an toàn (graceful shutdown)

Mỗi lần deploy bản mới, bản cũ bị dừng. Nếu dừng ngay lập tức, những request đang xử lý dở (một người đang nộp bài, một giao dịch đang ghi) bị cắt ngang.

actors: [ext] Nền tảng chạy, [go] Server cũ, [fe] Request đang chạy Nền tảng chạy -> Server cũ: SIGTERM (bản mới đã sẵn sàng) note Server cũ: ngừng nhận kết nối mới, load balancer chuyển sang bản mới Request đang chạy --> Server cũ: xử lý nốt, trả response Server cũ -> Server cũ: hết request → đóng database Nền tảng chạy -> Server cũ: quá thời gian chờ thì SIGKILL

Kubernetes gửi SIGTERM, chờ mặc định 30 giây (terminationGracePeriodSeconds), rồi mới gửi SIGKILL. Docker stop chờ 10 giây. Vì vậy thời gian chờ Shutdown trong code (20 giây ở trên) phải ngắn hơn thời gian chờ của nền tảng.

- Method và đường dẫn + Status code và thời gian xử lý - Request ID ? Status và thời gian chỉ biết sau khi handler chạy xong, nên phải ghi log sau next.ServeHTTP (hoặc trong defer).

Tổ chức lại API ghi chú theo lớp

Lấy API ghi chú của bài 11 (bản có GET /notes, POST /notes, GET /notes/{id}), tổ chức lại thành dự án nhiều package:

  • internal/notes: service (kiểm tra nội dung) + store lưu trong bộ nhớ, nối với nhau qua interface. Package này không import net/http.
  • internal/httpapi: handler + ba middleware RequestID, Logger, Recover.
  • cmd/api/main.go: đọc cổng từ biến môi trường ADDR, ghép mọi thứ, tắt server an toàn khi bấm Ctrl+C.
  • Thêm hai route để thử: GET /panic (phải nhận 500 và server vẫn chạy), GET /slow (chờ 5 giây rồi trả lời, để thử tắt an toàn).

Gợi ý từng bước

  1. Tạo khung: mkdir notes-api && cd notes-api && go mod init example.com/notes, rồi mkdir -p internal/notes internal/httpapi cmd/api.
  2. Viết internal/notes/notes.go trước: kiểu Note, lỗi ErrNotFound, ErrInvalid, interface Store, Service và MemStore. Viết test cho nó luôn (chưa cần HTTP).
  3. Viết internal/httpapi/middleware.go: chép ba middleware trong bài.
  4. Viết internal/httpapi/handler.go: handler chỉ đọc request, gọi service, đổi lỗi thành status ở một hàm writeError duy nhất. Hàm Routes() đăng ký đường dẫn và bọc middleware.
  5. Viết cmd/api/main.go: chép phần run() ở mục "Ghép mọi thứ trong main", bỏ phần database.
  6. Chạy: go run ./cmd/api (chú ý: chạy thư mục ./cmd/api, không phải ., vì main nằm ở đó).

// Package notes chứa nghiệp vụ ghi chú. Không import net/http.
package notes

import (
    "cmp"
    "context"
    "errors"
    "slices"
    "strings"
    "sync"
    "time"
    "unicode/utf8"
)

var (
    ErrNotFound = errors.New("không tìm thấy")
    ErrInvalid  = errors.New("nội dung từ 1 đến 500 ký tự")
)

type Note struct {
    ID        int64
    Text      string
    CreatedAt time.Time
}

// Store: nơi lưu ghi chú. Service chỉ biết interface này
type Store interface {
    Insert(ctx context.Context, n Note) (Note, error)
    Get(ctx context.Context, id int64) (Note, error)
    List(ctx context.Context) ([]Note, error)
}

type Service struct {
    store Store
    clock func() time.Time
}

func NewService(s Store) *Service {
    return &Service{store: s, clock: time.Now}
}

func (s *Service) Create(ctx context.Context, text string) (Note, error) {
    text = strings.TrimSpace(text)
    if n := utf8.RuneCountInString(text); n == 0 || n > 500 {
        return Note{}, ErrInvalid
    }
    return s.store.Insert(ctx, Note{Text: text, CreatedAt: s.clock()})
}

func (s *Service) Get(ctx context.Context, id int64) (Note, error) {
    return s.store.Get(ctx, id)
}

func (s *Service) List(ctx context.Context) ([]Note, error) {
    return s.store.List(ctx)
}

// MemStore: lưu trong bộ nhớ. Bài 15 thay bằng Postgres, Service không đổi
type MemStore struct {
    mu     sync.Mutex
    nextID int64
    notes  map[int64]Note
}

func NewMemStore() *MemStore {
    return &MemStore{notes: map[int64]Note{}}
}

func (m *MemStore) Insert(_ context.Context, n Note) (Note, error) {
    m.mu.Lock()
    defer m.mu.Unlock()
    m.nextID++
    n.ID = m.nextID
    m.notes[n.ID] = n
    return n, nil
}

func (m *MemStore) Get(_ context.Context, id int64) (Note, error) {
    m.mu.Lock()
    defer m.mu.Unlock()
    n, ok := m.notes[id]
    if !ok {
        return Note{}, ErrNotFound
    }
    return n, nil
}

func (m *MemStore) List(_ context.Context) ([]Note, error) {
    m.mu.Lock()
    defer m.mu.Unlock()
    out := make([]Note, 0, len(m.notes))
    for _, n := range m.notes {
        out = append(out, n)
    }
    slices.SortFunc(out, func(a, b Note) int { return cmp.Compare(a.ID, b.ID) })
    return out, nil
}

package httpapi

import (

    "encoding/hex"
    "log/slog"
    "net/http"
    "runtime/debug"
    "time"

type ctxKey string

const requestIDKey ctxKey = "request_id"

func RequestIDFrom(ctx context.Context) string {
    id, _ := ctx.Value(requestIDKey).(string)
    return id
}

// RequestID gắn mã cho mỗi request: vào ctx (để log) và vào header response (để người dùng báo lại)
func RequestID(next http.Handler) http.Handler {
    return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
        id := r.Header.Get("X-Request-ID")
        if id == "" || len(id) > 64 {
            b := make([]byte, 8)
            rand.Read(b)
            id = hex.EncodeToString(b)
        }
        w.Header().Set("X-Request-ID", id)
        ctx := context.WithValue(r.Context(), requestIDKey, id)
        next.ServeHTTP(w, r.WithContext(ctx))
    })
}

// statusRecorder nhớ status code mà handler đã ghi
type statusRecorder struct {
    http.ResponseWriter
    status int
}

func (s *statusRecorder) WriteHeader(code int) {
    s.status = code
    s.ResponseWriter.WriteHeader(code)
}

func (s *statusRecorder) Write(b []byte) (int, error) {
    if s.status == 0 {
        s.status = http.StatusOK
    }
    return s.ResponseWriter.Write(b)
}

// Logger ghi một dòng log cho mỗi request, SAU khi handler chạy xong
func Logger(log *slog.Logger) func(http.Handler) http.Handler {
    return func(next http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
            start := time.Now()
            rec := &statusRecorder{ResponseWriter: w}
            next.ServeHTTP(rec, r)
            log.Info("http",
                "method", r.Method,
                "path", r.URL.Path,
                "status", rec.status,
                "duration_ms", time.Since(start).Milliseconds(),
                "request_id", RequestIDFrom(r.Context()),
            )
        })
    }
}

// Recover chặn panic: trả 500, ghi log, server vẫn chạy
func Recover(log *slog.Logger) func(http.Handler) http.Handler {
    return func(next http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
            defer func() {
                if v := recover(); v != nil {
                    log.Error("panic", "value", v, "request_id", RequestIDFrom(r.Context()), "stack", string(debug.Stack()))
                    writeErr(w, http.StatusInternalServerError, "Có lỗi xảy ra, vui lòng thử lại")
                }
            }()
            next.ServeHTTP(w, r)
        })
    }
}

package httpapi

import (
    "encoding/json"
    "errors"
    "log/slog"
    "net/http"
    "strconv"
    "time"

    "example.com/notes/internal/notes"
)

type Handler struct {
    notes *notes.Service
    log   *slog.Logger
}

func New(svc *notes.Service, log *slog.Logger) *Handler {
    return &Handler{notes: svc, log: log}
}

// Routes trả về handler đã bọc đủ middleware: RequestID -> Logger -> Recover -> mux
func (h *Handler) Routes() http.Handler {
    mux := http.NewServeMux()
    mux.HandleFunc("GET /notes", h.list)
    mux.HandleFunc("POST /notes", h.create)
    mux.HandleFunc("GET /notes/{id}", h.get)
    mux.HandleFunc("GET /slow", func(w http.ResponseWriter, r *http.Request) {
        time.Sleep(5 * time.Second) // để thử tắt server an toàn
        writeJSON(w, http.StatusOK, map[string]string{"ok": "xong việc chậm"})
    })
    mux.HandleFunc("GET /panic", func(w http.ResponseWriter, r *http.Request) {
        panic("thử panic") // chỉ để thử middleware Recover
    })

    var handler http.Handler = mux
    handler = Recover(h.log)(handler) // lớp trong cùng
    handler = Logger(h.log)(handler)
    handler = RequestID(handler) // lớp ngoài cùng, chạy đầu tiên
    return handler
}

type noteDTO struct {
    ID        int64     `json:"id"`
    Text      string    `json:"text"`
    CreatedAt time.Time `json:"createdAt"`
}

func toDTO(n notes.Note) noteDTO { return noteDTO{ID: n.ID, Text: n.Text, CreatedAt: n.CreatedAt} }

func (h *Handler) list(w http.ResponseWriter, r *http.Request) {
    list, err := h.notes.List(r.Context())
    if err != nil {
        h.writeError(w, r, err)
        return
    }
    out := make([]noteDTO, 0, len(list))
    for _, n := range list {
        out = append(out, toDTO(n))
    }
    writeJSON(w, http.StatusOK, out)
}

func (h *Handler) create(w http.ResponseWriter, r *http.Request) {
    var req struct {
        Text string `json:"text"`
    }
    r.Body = http.MaxBytesReader(w, r.Body, 1<<16)
    if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
        writeErr(w, http.StatusBadRequest, "JSON không hợp lệ")
        return
    }
    n, err := h.notes.Create(r.Context(), req.Text)
    if err != nil {
        h.writeError(w, r, err)
        return
    }
    writeJSON(w, http.StatusCreated, toDTO(n))
}

func (h *Handler) get(w http.ResponseWriter, r *http.Request) {
    id, err := strconv.ParseInt(r.PathValue("id"), 10, 64)
    if err != nil {
        writeErr(w, http.StatusBadRequest, "id không hợp lệ")
        return
    }
    n, err := h.notes.Get(r.Context(), id)
    if err != nil {
        h.writeError(w, r, err)
        return
    }
    writeJSON(w, http.StatusOK, toDTO(n))
}

// writeError: chỗ DUY NHẤT đổi lỗi nghiệp vụ thành status code
func (h *Handler) writeError(w http.ResponseWriter, r *http.Request, err error) {
    switch {
    case errors.Is(err, notes.ErrInvalid):
        writeErr(w, http.StatusBadRequest, "Nội dung từ 1 đến 500 ký tự")
    case errors.Is(err, notes.ErrNotFound):
        writeErr(w, http.StatusNotFound, "Không tìm thấy")
    default:
        h.log.Error("lỗi không xử lý", "err", err, "request_id", RequestIDFrom(r.Context()))
        writeErr(w, http.StatusInternalServerError, "Có lỗi xảy ra, vui lòng thử lại")
    }
}

func writeJSON(w http.ResponseWriter, status int, v any) {
    w.Header().Set("Content-Type", "application/json; charset=utf-8")
    w.WriteHeader(status)
    json.NewEncoder(w).Encode(v)
}

func writeErr(w http.ResponseWriter, status int, msg string) {
    writeJSON(w, status, map[string]string{"error": msg})
}

package main

import (
    "context"
    "errors"
    "log/slog"
    "net/http"
    "os"
    "os/signal"
    "syscall"
    "time"

    "example.com/notes/internal/httpapi"
    "example.com/notes/internal/notes"
)

func main() {
    if err := run(); err != nil {
        slog.Error("dừng", "err", err)
        os.Exit(1)
    }
}

func run() error {
    addr := os.Getenv("ADDR")
    if addr == "" {
        addr = ":8080"
    }
    logger := slog.New(slog.NewJSONHandler(os.Stdout, nil))

    // Ghép các thành phần: store -> service -> handler
    svc := notes.NewService(notes.NewMemStore())
    h := httpapi.New(svc, logger)

    srv := &http.Server{
        Addr:              addr,
        Handler:           h.Routes(),
        ReadHeaderTimeout: 5 * time.Second,
        ReadTimeout:       15 * time.Second,
        WriteTimeout:      30 * time.Second,
        IdleTimeout:       60 * time.Second,
    }

    // ctx bị huỷ khi bấm Ctrl+C hoặc nhận SIGTERM
    ctx, stop := signal.NotifyContext(context.Background(), os.Interrupt, syscall.SIGTERM)
    defer stop()

    errCh := make(chan error, 1)
    go func() {
        logger.Info("bắt đầu", "addr", addr)
        if err := srv.ListenAndServe(); !errors.Is(err, http.ErrServerClosed) {
            errCh <- err
        }
    }()

    select {
    case err := <-errCh:
        return err
    case <-ctx.Done():
    }

    logger.Info("đang tắt, chờ các request dở dang")
    shutdownCtx, cancel := context.WithTimeout(context.Background(), 20*time.Second)
    defer cancel()
    if err := srv.Shutdown(shutdownCtx); err != nil {
        return err
    }
    logger.Info("đã tắt")
    return nil
}
