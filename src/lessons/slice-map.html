Array: độ dài cố định, ít khi dùng trực tiếp

package main

import "fmt"

func main() {
    var days [7]string         // mảng 7 chuỗi, mỗi phần tử ban đầu là ""
    scores := [3]int{8, 9, 10} // [3] = đúng 3 phần tử, không thêm bớt được

    days[0] = "Thứ Hai"
    fmt.Println(len(days), days[0])
    fmt.Println(len(scores), scores)
}

7 Thứ Hai
3 [8 9 10]

[3]int và [4]int là hai kiểu khác nhau. Gán mảng cho biến khác là sao chép toàn bộ. Vì cứng nhắc như vậy, code hằng ngày gần như chỉ dùng slice.

Slice: "mảng" bạn sẽ dùng hằng ngày

Slice giống mảng của JavaScript: thêm bớt được, độ dài thay đổi. Cách viết khác array đúng một chỗ: không ghi số trong [].

package main

import (
    "fmt"
    "slices"
)

func main() {
    names := []string{"An", "Bình"} // [] trống = slice. Đọc là "slice các string"
    names = append(names, "Chi")    // thêm vào cuối. PHẢI gán lại: names = append(...)
    fmt.Println(names, len(names))

    fmt.Println(names[0])   // lấy phần tử đầu, giống JS
    sub := names[1:3]       // cắt từ vị trí 1 đến TRƯỚC vị trí 3, giống names.slice(1, 3)
    fmt.Println(sub)

    fmt.Println(slices.Contains(names, "Chi")) // giống names.includes("Chi")

    nums := make([]int, 0, 100) // make(kiểu, len, cap): rỗng, nhưng giữ sẵn chỗ cho 100 phần tử
    fmt.Println(len(nums), cap(nums))
}

An
[Bình Chi]
true
0 100

Bên trong một slice

Hiểu phần này là tránh được một loại lỗi rất khó tìm. Một slice thực chất là một struct nhỏ gồm 3 trường: con trỏ tới một mảng nền, len (số phần tử đang dùng) và cap (sức chứa của mảng nền tính từ vị trí con trỏ).

Cạm bẫy 1: hai slice dùng chung bộ nhớ

package main

import "fmt"

func main() {
    a := []string{"A", "B", "C", "D", "E"}
    s := a[1:3] // ["B", "C"], nhưng KHÔNG sao chép: s nhìn vào chính mảng của a
    s[0] = "X"  // s[0] chính là a[1]
    fmt.Println(s)
    fmt.Println(a) // a cũng bị sửa!
}

[X C]
[A X C D E]

Trong JS, a.slice(1, 3) tạo mảng mới nên sửa nó không ảnh hưởng a. Trong Go, a[1:3] chỉ là cửa sổ nhìn vào cùng một vùng nhớ.

Cạm bẫy 2: append ghi đè dữ liệu của slice khác

Khi còn chỗ trong cap, append ghi thẳng vào mảng nền. Khi hết chỗ, nó cấp mảng mới lớn hơn (thường gấp đôi khi slice còn nhỏ), sao chép sang và trả về slice trỏ vào mảng mới. Vì vậy bạn không đoán được kết quả của append có dùng chung bộ nhớ với slice cũ hay không.

package main

import (
    "fmt"
    "slices"
)

func main() {
    // SAI: append vào một slice cắt ra từ slice khác
    a := []string{"A", "B", "C", "D", "E"}
    s := a[1:3]                   // len 2, cap 4 (từ vị trí 1 tới hết mảng a còn 4 chỗ)
    fmt.Println(len(s), cap(s))
    s = append(s, "Z")            // còn chỗ trong cap -> ghi thẳng vào a[3]
    fmt.Println(a)                // "D" bị ghi đè âm thầm

    // ĐÚNG: sao chép ra mảng riêng trước khi append
    b := []string{"A", "B", "C", "D", "E"}
    t := slices.Clone(b[1:3])     // t có vùng nhớ riêng, giống [...arr] trong JS
    t = append(t, "Z")
    fmt.Println(b)                // b không bị ảnh hưởng
    fmt.Println(t)
}

2 4
[A B C Z E]
[A B C D E]
[B C Z]

Luôn viết s = append(s, x). Quên gán lại thì phần tử mới có thể "biến mất" vì nó nằm trong mảng mới mà biến s không trỏ tới. Trình biên dịch sẽ báo lỗi nếu bạn gọi append mà bỏ kết quả.

Chuẩn bị sẵn sức chứa

Khi biết trước số phần tử (ví dụ đổi 1.000 dòng database thành 1.000 DTO), cấp sẵn cap để tránh cấp phát và sao chép nhiều lần:

out := make([]UserDTO, 0, len(users)) // len 0, cap = số người dùng
for _, u := range users {
    out = append(out, toDTO(u))       // không lần nào phải xin thêm bộ nhớ
}

Đây là trích đoạn: users, UserDTO, toDTO là những thứ có trong dự án thật (bài 12). Chỉ cần nhớ mẫu: biết trước số lượng thì make([]T, 0, n).

nil slice và slice rỗng trong JSON

package main

import (
    "encoding/json"
    "fmt"
)

func main() {
    var a []int  // khai báo mà không tạo: a là nil
    b := []int{} // tạo slice rỗng: không nil

    fmt.Println(len(a), len(b)) // len, append, range chạy bình thường với cả hai

    ja, _ := json.Marshal(a) // Marshal: đổi giá trị Go sang JSON, giống JSON.stringify
    jb, _ := json.Marshal(b)
    fmt.Println(string(ja)) // Marshal trả về []byte, đổi sang string để in
    fmt.Println(string(jb))
}

0 0
null
[]

Đây là lý do frontend đôi khi nhận "items": null thay vì [] và vỡ ở dòng items.map(...). Backend Go nên khởi tạo slice rỗng cho các trường danh sách trong response. Bây giờ bạn biết lỗi đó đến từ đâu.

Map: giống object / Map trong JavaScript

map[string]int đọc là "map có khoá kiểu string, giá trị kiểu int". Trong TS là Record<string, number>. Khác object JS: mọi khoá cùng một kiểu, mọi giá trị cùng một kiểu.

package main

import "fmt"

func main() {
    ages := map[string]int{
        "An":   20,
        "Bình": 22, // dấu phẩy cuối là BẮT BUỘC khi xuống dòng
    }
    ages["Chi"] = 19   // thêm (khoá chưa có) hoặc sửa (khoá đã có)
    delete(ages, "An") // xoá, giống delete obj.An
    fmt.Println(len(ages))

    // Khoá không có: trả về zero value (0), KHÔNG lỗi, KHÔNG undefined
    fmt.Println(ages["Không có"])

    // "comma ok": nhận thêm biến thứ hai cho biết khoá có tồn tại không
    age, ok := ages["Bình"]
    fmt.Println(age, ok)
    age, ok = ages["Dũng"]
    fmt.Println(age, ok) // 0 vì không có, ok = false

    counts := make(map[string]int) // map rỗng, ghi được ngay
    counts["go"]++                 // khoá chưa có thì bắt đầu từ 0, rồi tăng lên 1
    counts["go"]++
    fmt.Println(counts)
}

2
0
22 true
0 false
map[go:2]

Vì khoá không có cũng trả về 0, nên ages["Dũng"] == 0 không cho biết Dũng 0 tuổi hay không có Dũng. Khi cần phân biệt, luôn dùng dạng v, ok := m[k].

Ba điều phải nhớ về map

  1. Ghi vào nil map gây panic. var m map[string]int; m["a"] = 1 làm chương trình dừng. Luôn tạo bằng make hoặc map[...]...{}.
  2. Thứ tự duyệt là ngẫu nhiên, và Go cố tình làm ngẫu nhiên khác nhau mỗi lần chạy để bạn không vô tình phụ thuộc vào nó. Cần thứ tự thì lấy danh sách khoá ra rồi sắp xếp: 
    package main

import (
    "fmt"
    "maps"
    "slices"
)

func main() {
    ages := map[string]int{"Chi": 19, "An": 20, "Bình": 22}

    // maps.Keys lấy các khoá, slices.Sorted sắp xếp chúng (Go 1.23+)
    keys := slices.Sorted(maps.Keys(ages))
    for _, k := range keys {
        fmt.Println(k, ages[k])
    }
}

    An 20
Bình 22
Chi 19
  3. Map không an toàn khi nhiều goroutine cùng ghi. Runtime phát hiện được và dừng chương trình với lỗi fatal error: concurrent map writes. Trong HTTP server (mỗi request một goroutine), một map dùng chung phải được bảo vệ bằng mutex (bài 09).

Set: dùng map với giá trị rỗng

Go không có kiểu Set như JS. Người ta dùng map mà chỉ quan tâm khoá:

package main

import "fmt"

func main() {
    emails := []string{"lan@gmail.com", "an@gmail.com", "lan@gmail.com"}

    // struct{} là kiểu "rỗng", không tốn byte nào. struct{}{} là giá trị của kiểu đó
    seen := map[string]struct{}{}
    for _, email := range emails {
        if _, ok := seen[email]; ok { // đã gặp rồi
            fmt.Println("trùng:", email)
            continue // bỏ qua phần còn lại, sang vòng tiếp
        }
        seen[email] = struct{}{} // ghi nhận đã gặp
    }
    fmt.Println("số email khác nhau:", len(seen))
}

trùng: lan@gmail.com
số email khác nhau: 2

Dùng map[string]bool cũng được và dễ đọc hơn (if seen[email]). struct{} chỉ tiết kiệm bộ nhớ khi map rất lớn.

Map trong bộ nhớ là cache đơn giản nhất. Nhưng khi chạy 3 bản server sau load balancer, mỗi bản có map riêng: người dùng gọi lần một vào máy A, lần hai vào máy B và không thấy dữ liệu cũ. Đây là lý do các hệ thống thật dùng Redis cho cache dùng chung (bài 17).

Gom nhóm: mẫu bạn sẽ viết rất nhiều

Ví dụ có danh sách bài nộp, cần gom theo học sinh để tính số bài đã nộp:

package main

import "fmt"

// Một bài nộp: của học sinh nào, bao nhiêu điểm
type Submission struct {
    StudentID int64
    Score     float64
}

// Đếm số bài nộp của mỗi học sinh. Trả về map: mã học sinh -> số bài
func countByStudent(subs []Submission) map[int64]int {
    out := make(map[int64]int)
    for _, s := range subs {
        out[s.StudentID]++ // lần đầu gặp học sinh: 0 -> 1, lần sau tăng tiếp
    }
    return out
}

func main() {
    subs := []Submission{
        {StudentID: 1, Score: 8},
        {StudentID: 2, Score: 7},
        {StudentID: 1, Score: 9},
    }
    counts := countByStudent(subs)
    fmt.Println("học sinh 1:", counts[1], "bài")
    fmt.Println("học sinh 2:", counts[2], "bài")
    fmt.Println("học sinh 3:", counts[3], "bài") // không có: zero value 0
}

học sinh 1: 2 bài
học sinh 2: 1 bài
học sinh 3: 0 bài

Trong JS bạn sẽ viết subs.reduce((acc, s) => { acc[s.studentId] = (acc[s.studentId] ?? 0) + 1; return acc }, {}). Go viết bằng vòng for, và không cần ?? 0 vì khoá chưa có đã là 0.

Ở bài 16, bạn sẽ dùng chính mẫu này để sửa lỗi N+1 truy vấn: lấy hết dữ liệu liên quan trong một câu SQL rồi gom bằng map trong Go.

- [1 2 3 4] + [1 2 99 4] - [1 2 99 3 4] ? b có len 2 nhưng cap 4, còn chỗ trong mảng nền chung nên append ghi thẳng vào a[2]. - Chỉ một request được ghi, request kia bị bỏ qua - Không sao, map trong Go tự đồng bộ + Chương trình có thể dừng hẳn với lỗi fatal error: concurrent map writes ? Map không an toàn cho ghi đồng thời. Đây là lỗi fatal (không recover được), làm sập cả server. Dùng sync.Mutex hoặc sync.Map.

Bài tập

Đếm từ xuất hiện nhiều nhất

Viết topWords(text string, n int) []string trả về n từ xuất hiện nhiều nhất trong text, không phân biệt hoa thường. Với đoạn văn "Go nhanh. go gọn, GO dễ học và dễ đọc" và n = 2, kết quả là:

[go dễ]

(go xuất hiện 3 lần, dễ 2 lần. Để đơn giản, coi dấu câu dính vào từ: nhanh. là một từ.)

Gợi ý từng bước

  1. Đổi cả đoạn về chữ thường: strings.ToLower.
  2. Tách thành danh sách từ: strings.Fields(text) tách theo khoảng trắng, giống text.split(/\s+/).
  3. Đếm mỗi từ bằng map[string]int, giống ví dụ "Gom nhóm" ở trên.
  4. Map không sắp xếp được. Lấy các từ ra một slice, rồi sắp xếp slice đó theo số lần đếm, nhiều đứng trước.
  5. Lấy n phần tử đầu: words[:n] (nhớ kiểm tra slice có đủ n phần tử không).
Xem lời giải 
package main

import (
    "fmt"
    "slices"
    "strings"
)

func topWords(text string, n int) []string {
    // Bước 1-3: đếm từng từ
    counts := map[string]int{}
    for _, w := range strings.Fields(strings.ToLower(text)) {
        counts[w]++
    }

    // Bước 4: lấy các từ ra slice
    words := make([]string, 0, len(counts))
    for w := range counts { // range trên map, chỉ lấy khoá
        words = append(words, w)
    }

    // Sắp xếp. Hàm so sánh trả về số âm nếu a đứng trước b, số dương nếu a đứng sau,
    // giống hệt hàm truyền vào arr.sort((a, b) => ...) của JS
    slices.SortFunc(words, func(a, b string) int {
        if counts[a] != counts[b] {
            return counts[b] - counts[a] // a đếm nhiều hơn -> kết quả âm -> a đứng trước
        }
        return strings.Compare(a, b) // bằng nhau thì xếp theo chữ cái
    })

    // Bước 5: lấy n từ đầu
    if len(words) > n {
        words = words[:n]
    }
    return words
}

func main() {
    text := "Go nhanh. go gọn, GO dễ học và dễ đọc"
    fmt.Println(topWords(text, 2))
    fmt.Println(topWords(text, 100)) // n lớn hơn số từ: trả về tất cả, không lỗi
}

[go dễ]
[go dễ gọn, học nhanh. và đọc]

Giải thích: để ý dòng so sánh phụ theo chữ cái. Map duyệt theo thứ tự ngẫu nhiên, nên nếu không có dòng đó, các từ cùng xuất hiện 1 lần (gọn,, học...) sẽ đổi chỗ giữa các lần chạy, làm test lúc đúng lúc sai. Thử xoá dòng đó và chạy vài lần để thấy.

Phân trang trên slice

Danh sách có 7 học sinh, mỗi trang hiện 3 em. Viết hàm pageOf(items []string, page, size int) []string trả về các phần tử của trang page (trang đầu là 1). Không được làm chương trình dừng (panic) khi trang vượt quá dữ liệu hoặc tham số âm. Kết quả mong muốn:

trang 1: [An Bình Chi]
trang 2: [Dũng Em Giang]
trang 3: [Hà]
trang 4: []
trang 0: []

Gợi ý từng bước

  1. Trang page bắt đầu ở vị trí start = (page - 1) * size. Trang 2, mỗi trang 3: bắt đầu ở vị trí 3.
  2. Kết thúc ở start + size, nhưng không được vượt quá len(items). Hàm min(a, b) có sẵn từ Go 1.21.
  3. Trước khi cắt, kiểm tra: page hoặc size nhỏ hơn 1, hay start đã vượt quá dữ liệu, thì trả về slice rỗng.
Xem lời giải 
package main

import "fmt"

func pageOf(items []string, page, size int) []string {
    if page < 1 || size < 1 { // tham số vô nghĩa
        return []string{}
    }
    start := (page - 1) * size
    if start >= len(items) { // trang nằm sau dữ liệu
        return []string{}
    }
    end := min(start+size, len(items)) // trang cuối có thể thiếu phần tử
    return items[start:end]
}

func main() {
    kids := []string{"An", "Bình", "Chi", "Dũng", "Em", "Giang", "Hà"}
    for _, p := range []int{1, 2, 3, 4, 0} {
        fmt.Printf("trang %d: %v\n", p, pageOf(kids, p, 3))
    }
}

trang 1: [An Bình Chi]
trang 2: [Dũng Em Giang]
trang 3: [Hà]
trang 4: []
trang 0: []

Thử bỏ dòng kiểm tra start >= len(items) rồi chạy lại. Chương trình dừng với lỗi panic: runtime error: slice bounds out of range. JavaScript trả undefined hay mảng rỗng khi vượt chỉ số, còn Go thì dừng hẳn. Mọi chỉ số tính từ dữ liệu người dùng gửi lên (như ?page=999) đều phải kiểm tra như trên.

Dùng cho mọi kiểu: hàm trên chỉ nhận []string. Bài 06 dạy generics để viết một lần, dùng cho cả []int, []User: func pageOf[T any](items []T, page, size int) []T. Thân hàm giữ nguyên, chỉ đổi string thành T.

Truy cập ngoài phạm vi slice (items[10] khi len là 5) gây panic, khác với JavaScript trả undefined. Mọi chỉ số tính từ dữ liệu người dùng gửi lên đều phải kiểm tra như trên.
