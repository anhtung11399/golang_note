2. Lược đồ và câu truy vấn

-- +goose Up
CREATE TABLE classes (
    id         bigint GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    teacher_id bigint NOT NULL,
    name       text NOT NULL,
    capacity   int NOT NULL CHECK (capacity BETWEEN 1 AND 80),
    note       text,                                  -- được NULL
    created_at timestamptz NOT NULL DEFAULT now()
);
CREATE TABLE assignments (
    id        bigint GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    class_id  bigint NOT NULL REFERENCES classes(id) ON DELETE CASCADE,
    title     text NOT NULL,
    due_date  date NOT NULL
);
CREATE INDEX ON assignments (class_id, due_date);

-- +goose Down
DROP TABLE assignments;
DROP TABLE classes;

-- name: GetClass :one
SELECT * FROM classes
WHERE id = $1 AND teacher_id = $2;

-- name: ListClasses :many
SELECT id, name, capacity FROM classes
WHERE teacher_id = $1
ORDER BY created_at DESC
LIMIT $2 OFFSET $3;

-- name: CreateClass :one
INSERT INTO classes (teacher_id, name, capacity, note)
VALUES (@teacher_id, @name, @capacity, sqlc.narg(note))
RETURNING *;

-- name: RenameClass :execrows
UPDATE classes SET name = @name
WHERE id = @id AND teacher_id = @teacher_id;

-- name: ClassStats :many
SELECT c.id, c.name, count(a.id) AS assignment_count
FROM classes c
LEFT JOIN assignments a ON a.class_id = c.id
WHERE c.teacher_id = $1
GROUP BY c.id
ORDER BY c.name;

-- name: ClassesByIDs :many
SELECT id, name FROM classes WHERE id = ANY(@ids::bigint[]);

Chú thích -- name: TênHàm :kiểu báo cho sqlc tên hàm Go và kiểu kết quả:

KiểuHàm Go trả về
:onemột struct, không có dòng thì lỗi pgx.ErrNoRows
:manyslice struct
:execchỉ error
:execrows(int64, error): số dòng bị ảnh hưởng (để biết UPDATE có trúng không)
:copyfromchèn hàng loạt bằng lệnh COPY (rất nhanh)
:batchexec, :batchmanygửi nhiều câu trong một lượt mạng (pgx batch)

Tham số: $1 thì sqlc tự đặt tên theo cột. @name đặt tên rõ ràng. sqlc.narg(note) là tham số được phép NULL (thành *string). @ids::bigint[] với ANY là cách truyền danh sách, thay cho IN (...).

const getClass = `-- name: GetClass :one
SELECT id, teacher_id, name, capacity, note, created_at FROM classes
WHERE id = $1 AND teacher_id = $2
`

type GetClassParams struct {
    ID        int64 `json:"id"`
    TeacherID int64 `json:"teacher_id"`
}

func (q *Queries) GetClass(ctx context.Context, arg GetClassParams) (Class, error) {
    row := q.db.QueryRow(ctx, getClass, arg.ID, arg.TeacherID)
    var i Class
    err := row.Scan(&i.ID, &i.TeacherID, &i.Name, &i.Capacity, &i.Note, &i.CreatedAt)
    return i, err
}

type ClassStatsRow struct {
    ID              int64  `json:"id"`
    Name            string `json:"name"`
    AssignmentCount int64  `json:"assignment_count"`
}

4. Dùng trong store

package store

import (
    "context"
    "errors"

    "github.com/jackc/pgx/v5"
    "github.com/jackc/pgx/v5/pgxpool"

    "sqlcdemo/internal/db"
)

var ErrNotFound = errors.New("không tìm thấy")

type ClassStore struct {
    pool *pgxpool.Pool
    q    *db.Queries
}

func New(pool *pgxpool.Pool) *ClassStore {
    return &ClassStore{pool: pool, q: db.New(pool)}
}

func (s *ClassStore) Get(ctx context.Context, teacherID, id int64) (db.Class, error) {
    c, err := s.q.GetClass(ctx, db.GetClassParams{ID: id, TeacherID: teacherID})
    if errors.Is(err, pgx.ErrNoRows) {
        return db.Class{}, ErrNotFound // không để lỗi của pgx lọt ra ngoài store
    }
    return c, err
}

func (s *ClassStore) Rename(ctx context.Context, teacherID, id int64, name string) error {
    n, err := s.q.RenameClass(ctx, db.RenameClassParams{Name: name, ID: id, TeacherID: teacherID})
    if err != nil {
        return err
    }
    if n == 0 {
        return ErrNotFound // không phải lớp của giáo viên này, hoặc không tồn tại
    }
    return nil
}

// Transaction: q.WithTx(tx) cho một Queries chạy trong transaction
func (s *ClassStore) CreateWithFirstAssignment(ctx context.Context, teacherID int64, name, title string) error {
    tx, err := s.pool.Begin(ctx)
    if err != nil {
        return err
    }
    defer tx.Rollback(ctx) // đã Commit thì Rollback không làm gì

    q := s.q.WithTx(tx)
    c, err := q.CreateClass(ctx, db.CreateClassParams{TeacherID: teacherID, Name: name, Capacity: 30})
    if err != nil {
        return err
    }
    if _, err := q.CreateAssignment(ctx, db.CreateAssignmentParams{ClassID: c.ID, Title: title}); err != nil {
        return err
    }
    return tx.Commit(ctx)
}

Không trả thẳng struct db.Class ra JSON của API nếu nó có cột nội bộ (teacher_id, password_hash). Ở lớp handler, chép sang một struct DTO chỉ có các trường muốn lộ ra. Nhiều đội đặt luật: internal/db chỉ được import trong store.

5. Truy vấn động: lọc tuỳ chọn

Điểm yếu của sqlc: câu SQL cố định, không ghép thêm điều kiện lúc chạy được. Với bộ lọc tuỳ chọn (tìm theo tên nếu có, lọc theo môn nếu có), có ba cách:

-- Cách 1: tham số NULL được = bỏ qua điều kiện
-- name: SearchAssignments :many
SELECT * FROM assignments
WHERE class_id = @class_id
  AND (sqlc.narg(title)::text IS NULL OR title ILIKE '%' || sqlc.narg(title) || '%')
  AND (sqlc.narg(due_from)::date IS NULL OR due_date >= sqlc.narg(due_from))
ORDER BY due_date DESC
LIMIT 20;
  • Cách 1 (ở trên): gọn, đủ cho bảng vừa. Nhưng như bài {{n:toi-uu-query}} đã cảnh báo, (x IS NULL OR cột = x) có thể làm Postgres không dùng được index tốt. Luôn có ít nhất một điều kiện cố định có index (class_id) để thu hẹp trước.
  • Cách 2: viết vài câu riêng cho vài tổ hợp hay dùng.
  • Cách 3: với màn tìm kiếm thật sự động (quản trị, báo cáo), dùng bộ ghép SQL như squirrel hoặc goqu cho riêng chỗ đó, cùng project vẫn dùng sqlc cho phần còn lại.
import sq "github.com/Masterminds/squirrel"

psql := sq.StatementBuilder.PlaceholderFormat(sq.Dollar) // Postgres dùng $1, $2

q := psql.Select("id", "title", "due_date").From("assignments").Where(sq.Eq{"class_id": classID})
if f.Title != "" {
    q = q.Where(sq.ILike{"title": "%" + f.Title + "%"})
}
if !f.DueFrom.IsZero() {
    q = q.Where(sq.GtOrEq{"due_date": f.DueFrom})
}
sqlStr, args, err := q.OrderBy("due_date DESC").Limit(20).ToSql()
// SELECT id, title, due_date FROM assignments WHERE class_id = $1 AND title ILIKE $2 ORDER BY due_date DESC LIMIT 20
rows, err := pool.Query(ctx, sqlStr, args...)

Tên cột trong OrderBy do người dùng chọn (sắp xếp theo cột nào) không được truyền thẳng: đó vẫn là ghép chuỗi SQL. Đối chiếu với danh sách cho phép: map[string]string{"due": "due_date", "title": "title"}.

GORM

GORM là ORM: bạn khai báo struct, GORM tự sinh câu SQL. Rất nhiều dự án Go ở Việt Nam dùng GORM, nhất là hệ thống quản trị, CRUD nhiều bảng.

go get gorm.io/gorm gorm.io/driver/postgres

Model

type Class struct {
    ID          int64      `gorm:"primaryKey"`
    TeacherID   int64      `gorm:"not null;index"`
    Name        string     `gorm:"size:60;not null"`
    Capacity    int        `gorm:"not null;check:capacity BETWEEN 1 AND 80"`
    Note        *string    // con trỏ: được NULL
    Assignments []Assignment // quan hệ 1 - n (GORM đoán khoá ngoại ClassID)
    CreatedAt   time.Time  // GORM tự điền khi tạo
    UpdatedAt   time.Time  // GORM tự điền khi sửa
    DeletedAt   gorm.DeletedAt `gorm:"index"` // có trường này = "xoá mềm" (xem bẫy bên dưới)
}

type Assignment struct {
    ID      int64     `gorm:"primaryKey"`
    ClassID int64     `gorm:"not null;index"`
    Title   string    `gorm:"size:200;not null"`
    DueDate time.Time `gorm:"type:date;not null"`
}

Quy ước tên: struct Class → bảng classes, trường TeacherID → cột teacher_id. Đổi tên bảng: thêm method func (Class) TableName() string { return "lop_hoc" }.

Kết nối và các thao tác cơ bản

package main

import (
    "fmt"
    "log"
    "os"
    "time"

    "gorm.io/driver/postgres"
    "gorm.io/gorm"
    "gorm.io/gorm/logger"
)

// ... type Class, Assignment như trên

func main() {
    db, err := gorm.Open(postgres.Open(os.Getenv("DATABASE_URL")), &gorm.Config{
        Logger: logger.Default.LogMode(logger.Info), // IN mọi câu SQL GORM chạy: bật khi học / gỡ lỗi
    })
    if err != nil {
        log.Fatal(err)
    }
    sqlDB, _ := db.DB() // pool bên dưới, cấu hình như bài trước
    sqlDB.SetMaxOpenConns(20)

    // Tạo
    c := Class{TeacherID: 7, Name: "Lớp 5A", Capacity: 30}
    db.Create(&c) // c.ID được điền sau khi chèn
    fmt.Println("id mới:", c.ID)

    // Đọc một
    var got Class
    if err := db.First(&got, c.ID).Error; err != nil { // không có → gorm.ErrRecordNotFound
        log.Fatal(err)
    }

    // Đọc nhiều có điều kiện
    var list []Class
    db.Where("teacher_id = ? AND capacity >= ?", 7, 20).Order("name").Limit(20).Find(&list)

    // Sửa vài cột
    db.Model(&got).Updates(map[string]any{"name": "Lớp 5A1", "capacity": 32})

    // Xoá
    db.Delete(&got)
}

Năm cái bẫy của GORM

Bẫy 1: Updates bằng struct bỏ qua giá trị 0.

db.Model(&c).Updates(Class{Name: "5A", Capacity: 0}) // capacity KHÔNG được cập nhật
// UPDATE "classes" SET "name"='5A',"updated_at"=... WHERE id = 1

GORM coi 0, "", false là "không đặt". Muốn ghi giá trị 0, dùng map[string]any hoặc Select("capacity", "name").Updates(...).

Bẫy 2: N+1 khi đọc quan hệ.

var classes []Class
db.Find(&classes)
for i := range classes {
    db.Where("class_id = ?", classes[i].ID).Find(&classes[i].Assignments) // 1 câu mỗi lớp!
}

db.Preload("Assignments").Find(&classes)
// SELECT * FROM classes WHERE deleted_at IS NULL
// SELECT * FROM assignments WHERE class_id IN (1,2,3,...)   ← chỉ 2 câu

Preload chạy 2 câu. Joins("Teacher") chạy 1 câu JOIN (chỉ dùng cho quan hệ 1 - 1 hoặc n - 1).

Bẫy 3: Xoá mềm làm UNIQUE vướng. Có DeletedAt thì dòng "đã xoá" vẫn nằm trong bảng. Ràng buộc UNIQUE(email) sẽ chặn đăng ký lại cùng email. Sửa bằng index có điều kiện: CREATE UNIQUE INDEX ON users (email) WHERE deleted_at IS NULL. Muốn xoá thật: db.Unscoped().Delete(&c).

Bẫy 4: Quên kiểm tra lỗi. Mọi lệnh GORM trả về *gorm.DB, lỗi nằm ở .Error. Rất dễ viết db.Create(&c) rồi quên. Luôn viết if err := db.Create(&c).Error; err != nil.

Bẫy 5: Where bằng chuỗi ghép.

db.Where("name = '" + name + "'").Find(&list)   // SQL injection
db.Order(r.URL.Query().Get("sort")).Find(&list) // cũng là injection

db.Where("name = ?", name).Find(&list)          // tham số
db.Where(&Class{Name: name}).Find(&list)        // struct (bỏ qua giá trị rỗng)

Transaction và khoá dòng trong GORM

err := db.WithContext(ctx).Transaction(func(tx *gorm.DB) error {
    var c Class
    // SELECT ... FOR UPDATE: khoá dòng như bài trước
    if err := tx.Clauses(clause.Locking{Strength: "UPDATE"}).First(&c, classID).Error; err != nil {
        return err
    }
    var n int64
    tx.Model(&Enrollment{}).Where("class_id = ?", c.ID).Count(&n)
    if int(n) >= c.Capacity {
        return ErrClassFull // trả lỗi → GORM tự Rollback
    }
    return tx.Create(&Enrollment{ClassID: c.ID, StudentID: studentID}).Error // nil → Commit
})

Nhớ WithContext(ctx): không có thì truy vấn không bị huỷ khi client ngắt kết nối hoặc hết timeout.

Đừng dùng db.AutoMigrate(&Class{}) trên production. Nó chỉ thêm, không xoá, không đổi tên, và bạn không kiểm soát được thứ tự hay câu lệnh khoá bảng. Dùng file migration (phần dưới), AutoMigrate chỉ để thử nhanh trên máy.

Nhúng migration vào binary, chạy khi khởi động

package main

import (
    "database/sql"
    "embed"

    _ "github.com/jackc/pgx/v5/stdlib" // driver database/sql cho goose
    "github.com/pressly/goose/v3"
)

//go:embed migrations/*.sql
var migrations embed.FS

func migrate(dsn string) error {
    db, err := sql.Open("pgx", dsn)
    if err != nil {
        return err
    }
    defer db.Close()
    goose.SetBaseFS(migrations)
    if err := goose.SetDialect("postgres"); err != nil {
        return err
    }
    return goose.Up(db, "migrations") // goose tự giữ khoá trong Postgres: 3 bản server cùng khởi động cũng chỉ một bản chạy
}

Hai cách chạy migration khi deploy, chọn một:

  • Bước riêng trước deploy (nên dùng): CI chạy goose up rồi mới deploy code mới. Migration lỗi thì không deploy. Trên Kubernetes thường là một Job hoặc init container (bài {{n:kubernetes}}).
  • Khi server khởi động: tiện cho dự án nhỏ, nhưng migration chậm làm server khởi động lâu, health check có thể giết nó giữa chừng.

Bài tập

Viết store bài tập bằng sqlc

Dùng lược đồ ở trên. Viết các truy vấn sqlc và hàm store:

  • CreateAssignment(classID, title, dueDate)
  • ListDue(classID, from, to): bài có hạn trong khoảng, sắp theo hạn.
  • DeleteAssignment(teacherID, id): chỉ xoá được bài thuộc lớp của giáo viên đó, không trúng thì trả ErrNotFound.
  • CountByClass(classIDs []int64): số bài của nhiều lớp trong một câu.

Gợi ý từng bước

  1. Xoá có kiểm tra quyền: DELETE FROM assignments a USING classes c WHERE a.id = @id AND a.class_id = c.id AND c.teacher_id = @teacher_id, kiểu :execrows.
  2. Cột date với pgx sinh ra pgtype.Date. Thêm override db_type: date → go_type: time.Time trong sqlc.yaml để dùng time.Time.
  3. Nhiều lớp: WHERE class_id = ANY(@class_ids::bigint[]) GROUP BY class_id.
  4. Store đổi n == 0 thành ErrNotFound.

-- name: CreateAssignment :one
INSERT INTO assignments (class_id, title, due_date)
VALUES (@class_id, @title, @due_date)
RETURNING *;

-- name: ListDue :many
SELECT * FROM assignments
WHERE class_id = @class_id AND due_date BETWEEN @date_from AND @date_to
ORDER BY due_date, id;

-- name: DeleteAssignment :execrows
DELETE FROM assignments a
USING classes c
WHERE a.id = @id AND a.class_id = c.id AND c.teacher_id = @teacher_id;

-- name: CountByClass :many
SELECT class_id, count(*) AS n
FROM assignments
WHERE class_id = ANY(@class_ids::bigint[])
GROUP BY class_id;

import (
    "context"
    "time"

    "sqlcdemo/internal/db"

type AssignmentStore struct{ q *db.Queries }

func (s *AssignmentStore) Create(ctx context.Context, classID int64, title string, due time.Time) (db.Assignment, error) {
    return s.q.CreateAssignment(ctx, db.CreateAssignmentParams{ClassID: classID, Title: title, DueDate: due})
}

func (s *AssignmentStore) ListDue(ctx context.Context, classID int64, from, to time.Time) ([]db.Assignment, error) {
    return s.q.ListDue(ctx, db.ListDueParams{ClassID: classID, DateFrom: from, DateTo: to})
}

func (s *AssignmentStore) Delete(ctx context.Context, teacherID, id int64) error {
    n, err := s.q.DeleteAssignment(ctx, db.DeleteAssignmentParams{ID: id, TeacherID: teacherID})
    if err != nil {
        return err
    }
    if n == 0 {
        return ErrNotFound
    }
    return nil
}

// Trả map để lớp không có bài nào vẫn tra được (giá trị 0)
func (s *AssignmentStore) CountByClass(ctx context.Context, classIDs []int64) (map[int64]int64, error) {
    rows, err := s.q.CountByClass(ctx, classIDs)
    if err != nil {
        return nil, err
    }
    m := make(map[int64]int64, len(classIDs))
    for _, r := range rows {
        m[r.ClassID] = r.N
    }
    return m, nil
}

Giáo viên khác gọi xoá bài không phải của mình thì 0 dòng bị ảnh hưởng, store trả ErrNotFound, handler trả 404 (không nói "bài có tồn tại nhưng không phải của bạn"). CountByClass chỉ chạy một câu dù có 50 lớp, tránh N+1.

Tìm và sửa lỗi trong đoạn code GORM

Đoạn code dưới đây có 4 lỗi (hiệu năng, đúng sai, bảo mật). Tìm và sửa.

func (h *H) GradeAll(c *gin.Context) {
    var subs []Submission
    h.db.Where("assignment_id = " + c.Param("id")).Find(&subs)

Gợi ý từng bước

  1. Dòng Where: tham số từ URL ghép vào chuỗi.
  2. Vòng lặp có một câu SELECT mỗi bài nộp.
  3. autoGrade có thể trả 0.
  4. Mọi lệnh đều không kiểm tra .Error, không có WithContext, và không kiểm tra bài tập có thuộc lớp của giáo viên đang đăng nhập không.
Xem lời giải 
func (h *H) GradeAll(c *gin.Context) {
    aid, err := strconv.ParseInt(c.Param("id"), 10, 64)
    if err != nil {
        apiErr(c, 400, "BAD_ID", "Mã bài không hợp lệ")
        return
    }
    teacherID := c.GetInt64("userID")
    db := h.db.WithContext(c.Request.Context()) // 4. truy vấn bị huỷ theo request

    // 4. quyền: bài phải thuộc lớp của giáo viên này
    var n int64
    if err := db.Model(&Assignment{}).
        Joins("JOIN classes ON classes.id = assignments.class_id").
        Where("assignments.id = ? AND classes.teacher_id = ?", aid, teacherID).
        Count(&n).Error; err != nil {
        apiErr(c, 500, "INTERNAL", "Lỗi máy chủ")
        return
    }
    if n == 0 {
        apiErr(c, 404, "NOT_FOUND", "Không tìm thấy bài")
        return
    }

    var subs []Submission
    err = db.Where("assignment_id = ?", aid). // 1. tham số, không ghép chuỗi
        Preload("Student").                    // 2. một câu cho mọi học sinh, không N+1
        Find(&subs).Error
    if err != nil {
        apiErr(c, 500, "INTERNAL", "Lỗi máy chủ")
        return
    }

    err = db.Transaction(func(tx *gorm.DB) error {
        for i := range subs {
            score := autoGrade(subs[i], subs[i].Student)
            // 3. map: điểm 0 vẫn được ghi
            if err := tx.Model(&subs[i]).Updates(map[string]any{"score": score, "graded": true}).Error; err != nil {
                return err
            }
            subs[i].Score, subs[i].Graded = score, true
        }
        return nil
    })
    if err != nil {
        apiErr(c, 500, "INTERNAL", "Lỗi máy chủ")
        return
    }
    c.JSON(200, subs)
}

Bốn lỗi: (1) SQL injection qua c.Param("id"). (2) N+1: 40 học sinh là 41 câu SELECT, Preload("Student") (cần trường Student Student trong model Submission) đưa về 2 câu. (3) Updates(struct) bỏ qua điểm 0. (4) Không kiểm lỗi, không gắn context, và giáo viên bất kỳ chấm được bài của lớp khác. Bọc vòng cập nhật trong transaction để không chấm được một nửa rồi lỗi. Với hàng trăm dòng, còn có thể gộp các UPDATE thành một câu UPDATE ... FROM (VALUES ...).
