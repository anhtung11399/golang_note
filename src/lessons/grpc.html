Vì sao các dịch vụ Go hay nói chuyện bằng gRPC

REST + JSON là lựa chọn mặc định cho API mà trình duyệt / app gọi. Nhưng khi một hệ thống tách thành nhiều dịch vụ (dịch vụ lớp học, dịch vụ chấm điểm, dịch vụ thông báo), các dịch vụ gọi nhau hàng nghìn lần mỗi giây. Ở đó, nhiều công ty dùng gRPC:

  • Hợp đồng chặt: viết file .proto, sinh code cho cả server và client (Go, Java, Python, TypeScript...). Gọi hàm từ xa giống gọi hàm thường, sai kiểu là lỗi biên dịch.
  • Nhanh và nhỏ: dữ liệu mã hoá nhị phân bằng Protocol Buffers, nhỏ hơn JSON 3 đến 10 lần, mã hoá / giải mã nhanh hơn nhiều.
  • HTTP/2: nhiều lời gọi chạy song song trên một kết nối, có streaming hai chiều.
  • Có sẵn deadline (timeout lan truyền qua các dịch vụ), mã lỗi chuẩn, metadata (giống header), interceptor (giống middleware).

Giống tRPC hoặc GraphQL codegen ở chỗ: có một "nguồn sự thật" về kiểu dữ liệu, client và server sinh code từ đó. Khác ở chỗ gRPC không chạy thẳng trong trình duyệt (trình duyệt không cho điều khiển HTTP/2 đủ sâu). Muốn gọi từ web phải qua gRPC-Web, Connect, hoặc một gateway đổi sang REST.

Protocol Buffers

syntax = "proto3";

package class.v1;

option go_package = "sobaitap/gen/class/v1;classv1";

import "google/protobuf/timestamp.proto";

// Dịch vụ: danh sách các hàm gọi từ xa
service ClassService {
  rpc GetClass(GetClassRequest) returns (GetClassResponse);
  rpc ListStudents(ListStudentsRequest) returns (stream Student); // server gửi nhiều tin
}

message GetClassRequest {
  int64 id = 1;
}

message GetClassResponse {
  Class class = 1;
}

message Class {
  int64 id = 1;
  string name = 2;
  int32 capacity = 3;
  google.protobuf.Timestamp created_at = 4;
  optional string note = 5;       // optional: phân biệt "không có" với chuỗi rỗng
  repeated string subjects = 6;   // repeated: danh sách
  ClassStatus status = 7;
}

enum ClassStatus {
  CLASS_STATUS_UNSPECIFIED = 0;   // giá trị 0 luôn là "chưa đặt"
  CLASS_STATUS_ACTIVE = 1;
  CLASS_STATUS_LOCKED = 2;
}

message ListStudentsRequest {
  int64 class_id = 1;
}

message Student {
  int64 id = 1;
  string name = 2;
}

Những con số = 1, = 2 là số hiệu trường, không phải giá trị. Trên đường truyền, Protobuf gửi số hiệu thay vì tên trường, nên tin nhắn rất nhỏ. Hệ quả quan trọng:

ViệcAn toàn?
Thêm trường mới với số hiệu mớian toàn: bên cũ bỏ qua trường lạ, bên mới thấy giá trị mặc định khi bên cũ không gửi
Đổi tên trườngan toàn trên đường truyền (chỉ số hiệu đi qua mạng), nhưng làm hỏng code đang dùng tên cũ
Đổi số hiệu, đổi kiểukhông: bên cũ đọc sai dữ liệu
Xoá trườngđược, nhưng phải ghi reserved 5; reserved "note"; để không ai dùng lại số đó

Kiểu Protobuf sang Go: int64 → int64, string → string, repeated string → []string, optional string → *string, message → con trỏ struct (*Class), Timestamp → *timestamppb.Timestamp (đổi bằng .AsTime() và timestamppb.New(t)).

Server

package main

import (
    "context"
    "log"
    "net"
    "time"

    "google.golang.org/grpc"
    "google.golang.org/grpc/codes"
    "google.golang.org/grpc/status"
    "google.golang.org/protobuf/types/known/timestamppb"

    classv1 "sobaitap/gen/class/v1"
)

type classServer struct {
    classv1.UnimplementedClassServiceServer // bắt buộc nhúng: thêm rpc mới vào .proto không làm vỡ build
}

func (s *classServer) GetClass(ctx context.Context, req *classv1.GetClassRequest) (*classv1.GetClassResponse, error) {
    if req.GetId() <= 0 {
        return nil, status.Error(codes.InvalidArgument, "id phải lớn hơn 0")
    }
    if req.GetId() != 42 {
        return nil, status.Errorf(codes.NotFound, "không có lớp %d", req.GetId())
    }
    return &classv1.GetClassResponse{Class: &classv1.Class{
        Id:        42,
        Name:      "Lớp 5A",
        Capacity:  30,
        CreatedAt: timestamppb.New(time.Date(2026, 9, 5, 0, 0, 0, 0, time.UTC)),
        Subjects:  []string{"Toán", "Tiếng Việt"},
        Status:    classv1.ClassStatus_CLASS_STATUS_ACTIVE,
    }}, nil
}

func (s *classServer) ListStudents(req *classv1.ListStudentsRequest, stream grpc.ServerStreamingServer[classv1.Student]) error {
    names := []string{"An", "Bình", "Chi"}
    for i, n := range names {
        if err := stream.Context().Err(); err != nil {
            return err // client đã huỷ
        }
        if err := stream.Send(&classv1.Student{Id: int64(i + 1), Name: n}); err != nil {
            return err
        }
    }
    return nil // return = kết thúc stream
}

func main() {
    lis, err := net.Listen("tcp", ":50051")
    if err != nil {
        log.Fatal(err)
    }
    srv := grpc.NewServer()
    classv1.RegisterClassServiceServer(srv, &classServer{})
    log.Println("gRPC ở :50051")
    log.Fatal(srv.Serve(lis))
}

Luôn dùng các hàm req.GetId() thay vì req.Id: hàm Get trả giá trị mặc định khi req là nil, không bao giờ panic.

Client

package main

import (
    "context"
    "fmt"
    "io"
    "log"
    "time"

    "google.golang.org/grpc"
    "google.golang.org/grpc/credentials/insecure"
    "google.golang.org/grpc/status"

    classv1 "sobaitap/gen/class/v1"
)

func main() {
    // Tạo MỘT lần và dùng lại cho cả chương trình (an toàn khi gọi từ nhiều goroutine)
    conn, err := grpc.NewClient("localhost:50051", grpc.WithTransportCredentials(insecure.NewCredentials()))
    if err != nil {
        log.Fatal(err)
    }
    defer conn.Close()
    c := classv1.NewClassServiceClient(conn)

    // Luôn đặt deadline cho mỗi lời gọi
    ctx, cancel := context.WithTimeout(context.Background(), 2*time.Second)
    defer cancel()

    res, err := c.GetClass(ctx, &classv1.GetClassRequest{Id: 42})
    if err != nil {
        log.Fatal(err)
    }
    cl := res.GetClass()
    fmt.Println("lớp:", cl.GetName(), cl.GetCapacity(), cl.GetSubjects(), cl.GetCreatedAt().AsTime().Format("02/01/2006"))

    _, err = c.GetClass(ctx, &classv1.GetClassRequest{Id: 7})
    st := status.Convert(err)
    fmt.Println("lỗi:", st.Code(), "-", st.Message())

    stream, err := c.ListStudents(ctx, &classv1.ListStudentsRequest{ClassId: 42})
    if err != nil {
        log.Fatal(err)
    }
    for {
        s, err := stream.Recv()
        if err == io.EOF {
            break // server đã gửi xong
        }
        if err != nil {
            log.Fatal(err)
        }
        fmt.Println("học sinh:", s.GetId(), s.GetName())
    }
}

// Đổi lỗi nghiệp vụ → mã gRPC, ở một chỗ duy nhất (giống writeErr của REST)
func toStatus(err error) error {
    switch {
    case errors.Is(err, service.ErrNotFound):
        return status.Error(codes.NotFound, "không tìm thấy")
    case errors.Is(err, service.ErrClassLocked):
        return status.Error(codes.FailedPrecondition, "lớp đang bị khoá")
    case errors.Is(err, context.DeadlineExceeded):
        return status.Error(codes.DeadlineExceeded, "quá thời gian")
    default:
        slog.Error("lỗi nội bộ", "err", err)
        return status.Error(codes.Internal, "lỗi máy chủ") // không lộ chi tiết
    }
}

Deadline lan truyền

Gateway nhận request với timeout 2 giây, gọi Class service, Class service gọi Grading service. Deadline được gửi kèm mỗi lời gọi (header grpc-timeout) và trừ dần thời gian đã dùng. Nếu gateway đã hết giờ, các dịch vụ phía sau cũng dừng, không làm việc vô ích.

Lời gọi gRPC không có timeout mặc định. Quên đặt deadline là một lời gọi có thể treo mãi, giữ goroutine và kết nối. Quy tắc: client luôn tạo ctx có timeout, server luôn truyền ctx của request xuống mọi lời gọi tiếp theo.

Metadata và interceptor

Metadata là cặp khoá - giá trị đi kèm lời gọi, giống header HTTP: token, request id, ngôn ngữ.

Interceptor là middleware của gRPC: bọc mọi lời gọi để ghi log, đo thời gian, xác thực, bắt panic.

// Server: interceptor xác thực + ghi log cho mọi lời gọi unary
func authInterceptor(verify func(string) (int64, error)) grpc.UnaryServerInterceptor {
    return func(ctx context.Context, req any, info *grpc.UnaryServerInfo, handler grpc.UnaryHandler) (any, error) {
        md, _ := metadata.FromIncomingContext(ctx)
        tokens := md.Get("authorization") // khoá metadata luôn là chữ thường
        if len(tokens) == 0 {
            return nil, status.Error(codes.Unauthenticated, "thiếu token")
        }
        uid, err := verify(strings.TrimPrefix(tokens[0], "Bearer "))
        if err != nil {
            return nil, status.Error(codes.Unauthenticated, "token không hợp lệ")
        }
        ctx = context.WithValue(ctx, userIDKey{}, uid)

        start := time.Now()
        res, err := handler(ctx, req) // gọi hàm thật
        slog.Info("grpc", "method", info.FullMethod, "code", status.Code(err), "dur", time.Since(start))
        return res, err
    }
}

srv := grpc.NewServer(
    grpc.ChainUnaryInterceptor(recoverInterceptor, authInterceptor(verifyToken)),
)

// Client: gắn token vào mọi lời gọi
ctx = metadata.AppendToOutgoingContext(ctx, "authorization", "Bearer "+token)
res, err := c.GetClass(ctx, req)

Không cần tự viết hết: github.com/grpc-ecosystem/go-grpc-middleware/v2 có sẵn interceptor log (slog), recovery, retry, rate limit. OpenTelemetry có otelgrpc tự gắn trace (bài {{n:quan-sat}}).

Những thứ cần cho production

  • Health check chuẩn: google.golang.org/grpc/health. Kubernetes có probe gRPC gọi thẳng dịch vụ này.
  • Tắt êm: srv.GracefulStop() khi nhận SIGTERM: ngừng nhận lời gọi mới, chờ lời gọi đang chạy xong.
  • TLS: giữa các dịch vụ dùng mTLS (cả hai bên có chứng chỉ). Thường do service mesh (Istio, Linkerd) lo, code dùng insecure nói chuyện với sidecar.
  • Cân bằng tải: gRPC giữ một kết nối lâu dài. Load balancer tầng 4 (TCP) chỉ chia kết nối, nên mọi lời gọi của một client dồn về một server. Dùng load balancer hiểu HTTP/2 (Envoy, ALB có hỗ trợ gRPC) hoặc cân bằng phía client (dns:/// + round_robin, headless Service trên Kubernetes).
  • Kích thước tin: mặc định tối đa 4MB. File lớn thì dùng client streaming, hoặc tốt hơn: upload thẳng lên kho file (bài {{n:realtime-file}}) và chỉ gửi khoá file.
// Cân bằng tải phía client qua DNS (trên K8s dùng headless Service để DNS trả về IP mọi pod)

Cho trình duyệt gọi: gateway và Connect

Hai cách phổ biến để cùng một dịch vụ phục vụ cả gRPC (nội bộ) và JSON (trình duyệt):

  • grpc-gateway: thêm chú thích HTTP vào .proto, sinh ra một reverse proxy đổi GET /v1/classes/42 thành lời gọi gRPC. Sinh luôn được file OpenAPI.
  • Connect (connectrpc.com/connect): thư viện thay thế grpc-go, một handler net/http nói được cả gRPC, gRPC-Web và Connect (JSON qua HTTP POST). Frontend dùng @connectrpc/connect-web sinh client TypeScript có kiểu.
import "google/api/annotations.proto";

service ClassService {
  rpc GetClass(GetClassRequest) returns (GetClassResponse) {
    option (google.api.http) = { get: "/v1/classes/{id}" };
  }
}

Ở các công ty dùng microservice (Grab, Uber, Tiki, VNG), mẫu hay gặp: gateway REST/GraphQL cho app và web, bên trong toàn gRPC. File .proto để trong một repo chung, CI chạy buf breaking và sinh thư viện client cho mọi ngôn ngữ.

- Ổn, Protobuf tự đổi kiểu + Không: dịch vụ cũ đọc sai dữ liệu. Thêm field mới với số hiệu mới, xoá field cũ và reserved số 5 - Ổn nếu deploy mọi dịch vụ cùng lúc ? Số hiệu và kiểu là hợp đồng trên đường truyền. Không thể deploy mọi dịch vụ đúng cùng một lúc, nên luôn có lúc bản cũ và bản mới nói chuyện với nhau. + Lời gọi chờ mãi, giữ goroutine và tài nguyên phía client, dồn lại có thể làm client cũng treo - Sau 30 giây gRPC tự huỷ - Server tự trả DeadlineExceeded ? gRPC không có timeout mặc định. Luôn dùng context.WithTimeout khi gọi. - Đều cho 5 pod + Gần như tất cả vào một pod, vì cả kết nối HTTP/2 được giữ với một pod - Ngẫu nhiên từng lời gọi ? Load balancer TCP chia theo kết nối, mà gRPC dùng một kết nối cho nhiều lời gọi. Cần load balancer hiểu HTTP/2 hoặc cân bằng phía client.

Bài tập

Dịch vụ chấm điểm gRPC có interceptor đo thời gian

Viết file grading.proto có rpc Grade(GradeRequest) returns (GradeResponse) (nhận submission_id, score 0 đến 10, comment tối đa 500 ký tự) và rpc WatchGrades(WatchRequest) returns (stream GradeEvent) (đẩy sự kiện mỗi khi có bài được chấm trong một lớp). Viết server, một interceptor ghi log tên hàm, mã lỗi và thời gian, và một client.

Gợi ý từng bước

  1. Điểm có thể lẻ (8.5): dùng double hoặc lưu theo đơn vị 0,1 điểm bằng int32 (85). Kiểu nguyên tránh sai số làm tròn.
  2. Validate trong Grade, sai thì codes.InvalidArgument.
  3. Phát sự kiện tới các stream đang nghe: một map classID → []chan *GradeEvent có mutex. WatchGrades đăng ký một channel, vòng select giữa channel và stream.Context().Done(), thoát thì gỡ đăng ký.
  4. Interceptor stream có kiểu khác: grpc.StreamServerInterceptor. Bài này chỉ cần interceptor unary.

syntax = "proto3";
package grading.v1;
option go_package = "sobaitap/gen/grading/v1;gradingv1";

service GradingService {
  rpc Grade(GradeRequest) returns (GradeResponse);
  rpc WatchGrades(WatchRequest) returns (stream GradeEvent);
}

message GradeRequest {
  int64 class_id = 1;
  int64 submission_id = 2;
  int32 score_tenths = 3; // 85 = 8,5 điểm
  string comment = 4;
}
message GradeResponse {}
message WatchRequest { int64 class_id = 1; }
message GradeEvent {
  int64 submission_id = 1;
  int32 score_tenths = 2;
}

type hub struct {
    mu   sync.Mutex
    subs map[int64]map[chan *gradingv1.GradeEvent]struct{}
}

func (h *hub) add(classID int64) chan *gradingv1.GradeEvent {
    ch := make(chan *gradingv1.GradeEvent, 16)
    h.mu.Lock()
    defer h.mu.Unlock()
    if h.subs[classID] == nil {
        h.subs[classID] = map[chan *gradingv1.GradeEvent]struct{}{}
    }
    h.subs[classID][ch] = struct{}{}
    return ch
}

func (h *hub) remove(classID int64, ch chan *gradingv1.GradeEvent) {
    h.mu.Lock()
    defer h.mu.Unlock()
    delete(h.subs[classID], ch)
}

func (h *hub) publish(classID int64, ev *gradingv1.GradeEvent) {
    h.mu.Lock()
    defer h.mu.Unlock()
    for ch := range h.subs[classID] {
        select {
        case ch <- ev:
        default: // người nghe chậm, đầy bộ đệm → bỏ tin, không làm chậm người chấm
        }
    }
}

type server struct {
    gradingv1.UnimplementedGradingServiceServer
    hub *hub
}

func (s *server) Grade(ctx context.Context, r *gradingv1.GradeRequest) (*gradingv1.GradeResponse, error) {
    if r.GetScoreTenths() < 0 || r.GetScoreTenths() > 100 {
        return nil, status.Error(codes.InvalidArgument, "điểm từ 0 đến 10")
    }
    if utf8.RuneCountInString(r.GetComment()) > 500 {
        return nil, status.Error(codes.InvalidArgument, "nhận xét tối đa 500 ký tự")
    }
    // ... lưu vào database
    s.hub.publish(r.GetClassId(), &gradingv1.GradeEvent{SubmissionId: r.GetSubmissionId(), ScoreTenths: r.GetScoreTenths()})
    return &gradingv1.GradeResponse{}, nil
}

func (s *server) WatchGrades(r *gradingv1.WatchRequest, st grpc.ServerStreamingServer[gradingv1.GradeEvent]) error {
    ch := s.hub.add(r.GetClassId())
    defer s.hub.remove(r.GetClassId(), ch)
    for {
        select {
        case ev := <-ch:
            if err := st.Send(ev); err != nil {
                return err
            }
        case <-st.Context().Done():
            return nil // client ngắt
        }
    }
}

func timing(ctx context.Context, req any, info *grpc.UnaryServerInfo, h grpc.UnaryHandler) (any, error) {
    start := time.Now()
    res, err := h(ctx, req)
    slog.Info("grpc", "method", info.FullMethod, "code", status.Code(err).String(), "dur", time.Since(start).Round(time.Microsecond))
    return res, err
}

func main() {
    lis, err := net.Listen("tcp", ":50052")
    if err != nil {
        log.Fatal(err)
    }
    srv := grpc.NewServer(grpc.ChainUnaryInterceptor(timing))
    gradingv1.RegisterGradingServiceServer(srv, &server{hub: &hub{subs: map[int64]map[chan *gradingv1.GradeEvent]struct{}{}}})
    log.Fatal(srv.Serve(lis))
}

package main

import (
    "context"
    "fmt"
    "log"
    "time"

    "google.golang.org/grpc"
    "google.golang.org/grpc/credentials/insecure"
    "google.golang.org/grpc/status"

    gradingv1 "sobaitap/gen/grading/v1"
)

func main() {
    conn, err := grpc.NewClient("localhost:50052", grpc.WithTransportCredentials(insecure.NewCredentials()))
    if err != nil {
        log.Fatal(err)
    }
    defer conn.Close()
    c := gradingv1.NewGradingServiceClient(conn)

    ctx, cancel := context.WithTimeout(context.Background(), 3*time.Second)
    defer cancel()

    stream, _ := c.WatchGrades(ctx, &gradingv1.WatchRequest{ClassId: 42})
    time.Sleep(100 * time.Millisecond) // chờ đăng ký xong (chỉ cho ví dụ)

    c.Grade(ctx, &gradingv1.GradeRequest{ClassId: 42, SubmissionId: 1, ScoreTenths: 85})
    c.Grade(ctx, &gradingv1.GradeRequest{ClassId: 42, SubmissionId: 2, ScoreTenths: 100})
    _, err = c.Grade(ctx, &gradingv1.GradeRequest{ClassId: 42, SubmissionId: 3, ScoreTenths: 120})
    fmt.Println("chấm sai:", status.Code(err), status.Convert(err).Message())

    for range 2 {
        ev, err := stream.Recv()
        if err != nil {
            log.Fatal(err)
        }
        fmt.Printf("sự kiện: bài %d được %.1f điểm\n", ev.GetSubmissionId(), float64(ev.GetScoreTenths())/10)
    }
}
