Driver và connection pool

Go có giao diện chung database/sql trong thư viện chuẩn, nhưng với PostgreSQL, thư viện được dùng nhiều nhất là pgx (github.com/jackc/pgx/v5): nhanh hơn, hỗ trợ đầy đủ kiểu dữ liệu của Postgres, có sẵn pool kết nối.

go get github.com/jackc/pgx/v5

import "github.com/jackc/pgx/v5/pgxpool"

pool, err := pgxpool.New(ctx, "postgres://app:secret@localhost:5432/sobaitap?sslmode=disable")
if err != nil {
    return err
}
defer pool.Close()

if err := pool.Ping(ctx); err != nil { // kiểm tra kết nối được ngay lúc khởi động
    return fmt.Errorf("không kết nối được database: %w", err)
}

Pool là gì? Mở một kết nối tới Postgres tốn thời gian (bắt tay TCP, TLS, xác thực) và mỗi kết nối chiếm bộ nhớ phía database. Pool giữ sẵn một nhóm kết nối, mỗi truy vấn mượn một cái rồi trả lại.

[go] 500 goroutine / request đang chạy -> mượn / trả [cache] pgxpool / tối đa N kết nối -> N kết nối TCP [db] PostgreSQL / max_connections mặc định 100

Kích thước pool không phải càng to càng tốt. Postgres xử lý mỗi kết nối bằng một tiến trình riêng, nên vài trăm kết nối đồng thời làm nó chậm đi. Nếu chạy 10 bản server, mỗi bản pool 20 kết nối, tổng là 200, vượt max_connections mặc định (100). Đặt qua URL: ?pool_max_conns=10. Hệ thống lớn đặt PgBouncer ở giữa để gom kết nối.

Chạy thử: chương trình Go đầu tiên đọc database

Trước khi đi vào từng phần, chạy một chương trình hoàn chỉnh để thấy toàn cảnh. Cần có Postgres đang chạy, đã tạo 5 bảng và nạp dữ liệu mẫu ở bài 14.

Bước 1. Tạo dự án và cài pgx:

mkdir hello-db && cd hello-db
go mod init hello-db
go get github.com/jackc/pgx/v5   # tải thư viện, ghi vào go.mod

Bước 2. Tạo main.go:

package main

import (
    "context"
    "errors"
    "fmt"
    "os"

    "github.com/jackc/pgx/v5"
    "github.com/jackc/pgx/v5/pgconn"
    "github.com/jackc/pgx/v5/pgxpool"
)

type Class struct {
    ID       int64
    Name     string
    Capacity int
}

func main() {
    ctx := context.Background()

    // 1. Mở pool kết nối. Địa chỉ database lấy từ biến môi trường, không ghi cứng trong code
    pool, err := pgxpool.New(ctx, os.Getenv("DATABASE_URL"))
    if err != nil {
        fmt.Println("địa chỉ database sai:", err)
        os.Exit(1)
    }
    defer pool.Close()
    if err := pool.Ping(ctx); err != nil {
        fmt.Println("không kết nối được database:", err)
        os.Exit(1)
    }

    // 2. Đọc MỘT dòng: QueryRow + Scan
    var name string
    err = pool.QueryRow(ctx, `SELECT name FROM users WHERE email = $1`, "an@hs.vn").Scan(&name)
    fmt.Println("tìm an@hs.vn:", name, err)

    err = pool.QueryRow(ctx, `SELECT name FROM users WHERE email = $1`, "khongco@hs.vn").Scan(&name)
    fmt.Println("tìm khongco@hs.vn: không thấy?", errors.Is(err, pgx.ErrNoRows))

    // 3. Đọc NHIỀU dòng: Query + vòng lặp rows.Next
    rows, err := pool.Query(ctx, `SELECT id, name, capacity FROM classes WHERE teacher_id = $1 ORDER BY id`, 7)
    if err != nil {
        fmt.Println("lỗi truy vấn:", err)
        os.Exit(1)
    }
    classes := []Class{}
    for rows.Next() {
        var c Class
        if err := rows.Scan(&c.ID, &c.Name, &c.Capacity); err != nil {
            fmt.Println("lỗi đọc dòng:", err)
            os.Exit(1)
        }
        classes = append(classes, c)
    }
    rows.Close()
    fmt.Printf("lớp của giáo viên 7: %+v\n", classes)

    // 4. Ghi: Exec trả về số dòng bị ảnh hưởng
    tag, err := pool.Exec(ctx, `UPDATE classes SET name = $1 WHERE id = $2 AND teacher_id = $3`, "Lớp 5A1", 42, 7)
    fmt.Println("đổi tên lớp 42:", tag.RowsAffected(), "dòng", err)
    tag, err = pool.Exec(ctx, `UPDATE classes SET name = $1 WHERE id = $2 AND teacher_id = $3`, "Của tôi", 50, 7)
    fmt.Println("đổi tên lớp 50 (của người khác):", tag.RowsAffected(), "dòng", err)

    // 5. Lỗi ràng buộc: đọc mã lỗi của Postgres
    _, err = pool.Exec(ctx, `INSERT INTO users (email, name, role) VALUES ($1, $2, $3)`, "an@hs.vn", "An 2", "student")
    var pgErr *pgconn.PgError
    if errors.As(err, &pgErr) {
        fmt.Println("thêm email trùng: mã lỗi", pgErr.Code, "| ràng buộc", pgErr.ConstraintName)
    }
}

Gặp không kết nối được database: ... connection refused: Postgres chưa chạy (mở Docker Desktop, hoặc brew services start postgresql@17). Gặp relation "users" does not exist: chưa tạo bảng ở bài 14.

Các phần dưới giải thích từng việc trong chương trình trên: QueryRow đọc một dòng, Query đọc nhiều dòng, Exec ghi, và cách đọc lỗi của Postgres. Các đoạn code bên dưới là trích đoạn từ một Store (bài 13), cách viết thật trong dự án.

Truy vấn có tham số: không bao giờ ghép chuỗi

// SQL injection: email = "x' OR '1'='1" là lấy được mọi người dùng
q := "SELECT id, name FROM users WHERE email = '" + email + "'"
rows, err := pool.Query(ctx, q)

// $1, $2... là chỗ giữ chỗ. Giá trị được gửi RIÊNG, database không bao giờ coi nó là code SQL
row := pool.QueryRow(ctx, `SELECT id, name FROM users WHERE email = $1`, email)

Tham số chỉ thay được giá trị, không thay được tên bảng, tên cột hay ASC/DESC. Những phần đó nếu phải động thì chọn từ danh sách cố định trong code (whitelist, bài 12), không bao giờ lấy thẳng từ request.

Đọc một dòng

type User struct {
    ID        int64
    Email     string
    Name      string
    CreatedAt time.Time
}

func (s *Store) UserByEmail(ctx context.Context, email string) (User, error) {
    var u User
    err := s.db.QueryRow(ctx,
        `SELECT id, email, name, created_at FROM users WHERE email = $1`, email,
    ).Scan(&u.ID, &u.Email, &u.Name, &u.CreatedAt) // thứ tự Scan khớp thứ tự cột
    if errors.Is(err, pgx.ErrNoRows) {
        return User{}, domain.ErrNotFound
    }
    if err != nil {
        return User{}, fmt.Errorf("tìm user theo email: %w", err)
    }
    return u, nil
}

Đọc nhiều dòng

func (s *Store) ClassesOf(ctx context.Context, teacherID int64) ([]Class, error) {
    rows, err := s.db.Query(ctx,
        `SELECT id, name, capacity FROM classes WHERE teacher_id = $1 ORDER BY created_at DESC`, teacherID)
    if err != nil {
        return nil, err
    }
    defer rows.Close() // trả kết nối về pool

    out := []Class{} // không nil → JSON là [] chứ không phải null
    for rows.Next() {
        var c Class
        if err := rows.Scan(&c.ID, &c.Name, &c.Capacity); err != nil {
            return nil, err
        }
        out = append(out, c)
    }
    return out, rows.Err() // lỗi xảy ra giữa chừng lúc đọc (mất kết nối...)
}

pgx có hàm tiện ích giúp bớt code lặp:

rows, _ := s.db.Query(ctx, `SELECT id, name, capacity FROM classes WHERE teacher_id = $1`, teacherID)
classes, err := pgx.CollectRows(rows, pgx.RowToStructByName[Class]) // khớp tên cột với tên trường

Quên rows.Close() là rò kết nối: sau vài trăm request, pool cạn và mọi request treo ở bước chờ kết nối. Quên rows.Err() là có thể trả về danh sách thiếu mà không biết.

Ghi dữ liệu

func (s *Store) CreateClass(ctx context.Context, c Class) (Class, error) {
    err := s.db.QueryRow(ctx,
        `INSERT INTO classes (teacher_id, name, capacity) VALUES ($1, $2, $3)
         RETURNING id, created_at`,
        c.TeacherID, c.Name, c.Capacity,
    ).Scan(&c.ID, &c.CreatedAt)
    return c, err
}

func (s *Store) RenameClass(ctx context.Context, teacherID, classID int64, name string) error {
    tag, err := s.db.Exec(ctx,
        `UPDATE classes SET name = $1 WHERE id = $2 AND teacher_id = $3`, name, classID, teacherID)
    if err != nil {
        return err
    }
    if tag.RowsAffected() == 0 { // không có lớp đó, hoặc không phải lớp của giáo viên này
        return domain.ErrNotFound
    }
    return nil
}

Đổi lỗi ràng buộc thành lỗi nghiệp vụ

import "github.com/jackc/pgx/v5/pgconn"

_, err := s.db.Exec(ctx, `INSERT INTO users (email, name, role) VALUES ($1, $2, $3)`, email, name, role)
var pgErr *pgconn.PgError
if errors.As(err, &pgErr) && pgErr.Code == "23505" { // unique_violation
    return domain.ErrEmailTaken // → 409 "Email đã được dùng"
}

Đây là cách đúng để chống trùng email: không SELECT kiểm tra trước rồi mới INSERT (hai request cùng lúc đều thấy "chưa có"), mà cứ INSERT và để ràng buộc UNIQUE từ chối. Mã lỗi Postgres hay gặp: 23505 trùng UNIQUE, 23503 sai khoá ngoại, 23514 sai CHECK, 40001 xung đột tuần tự hoá (bài 16).

Transaction: tất cả hoặc không gì cả

Nộp bài gồm hai việc: lưu bài nộp và tạo thông báo cho giáo viên (bảng notifications trong ví dụ này chưa có ở bài 14, chỉ để minh hoạ). Nếu lưu xong bài mà lỗi lúc tạo thông báo thì sao? Transaction đảm bảo các câu lệnh bên trong cùng thành công hoặc cùng bị huỷ.

func (s *Store) Submit(ctx context.Context, sub Submission, teacherID int64) error {
    tx, err := s.db.Begin(ctx)
    if err != nil {
        return err
    }
    defer tx.Rollback(ctx) // nếu đã Commit thì Rollback không làm gì, an toàn

    if _, err := tx.Exec(ctx,
        `INSERT INTO submissions (assignment_id, student_id) VALUES ($1, $2)
         ON CONFLICT (assignment_id, student_id) DO UPDATE SET submitted_at = now()
         WHERE submissions.score IS NULL`, // đã chấm thì không cho nộp lại
        sub.AssignmentID, sub.StudentID); err != nil {
        return err
    }
    if _, err := tx.Exec(ctx,
        `INSERT INTO notifications (user_id, kind, ref_id) VALUES ($1, 'submitted', $2)`,
        teacherID, sub.AssignmentID); err != nil {
        return err
    }
    return tx.Commit(ctx)
}

pgx có sẵn hàm bọc gọn hơn: pgx.BeginFunc(ctx, pool, func(tx pgx.Tx) error { ... }) tự commit khi hàm trả nil và tự rollback khi lỗi.

Không gọi API bên ngoài (gửi email, gọi cổng thanh toán) bên trong transaction. Transaction giữ khoá trên các dòng đã sửa. Nếu dịch vụ bên ngoài mất 10 giây, các request khác chạm vào những dòng đó cũng phải chờ 10 giây. Ghi vào database trong transaction, việc gọi ra ngoài làm sau (hoặc qua hàng đợi, bài 20).

Migration: đổi cấu trúc database có kiểm soát

Lược đồ database thay đổi theo thời gian. Migration là các file SQL đánh số, chạy lần lượt và chỉ chạy một lần, được commit vào git cùng với code. Mọi máy (bạn, đồng nghiệp, CI, production) đều đi qua đúng cùng các bước.

Quy tắc vàng với production đang chạy: migration phải tương thích với cả code cũ lẫn code mới, vì trong lúc deploy có một khoảng cả hai phiên bản cùng chạy. Muốn đổi tên cột: thêm cột mới → code ghi cả hai cột → chép dữ liệu cũ sang → code chỉ đọc cột mới → bản deploy sau mới xoá cột cũ. Không bao giờ đổi tên hay xoá cột trong một bước.

Cẩn thận với bảng lớn: CREATE INDEX thường khoá ghi trên cả bảng trong lúc tạo. Trên production dùng CREATE INDEX CONCURRENTLY.

sqlc: viết SQL, nhận code Go có kiểu

Viết tay Scan cho hàng chục câu truy vấn vừa chán vừa dễ sai thứ tự cột. sqlc đọc file SQL và lược đồ của bạn, rồi sinh ra code Go có kiểu đầy đủ. Bạn vẫn viết SQL thật (không phải ORM), nhưng được trình biên dịch kiểm tra.

-- name: ClassesOfTeacher :many
SELECT id, name, capacity FROM classes
WHERE teacher_id = $1
ORDER BY created_at DESC;

-- name: CreateClass :one
INSERT INTO classes (teacher_id, name, capacity) VALUES ($1, $2, $3)
RETURNING *;

sqlc generate   # sinh ra classes.sql.go

q := db.New(pool)
classes, err := q.ClassesOfTeacher(ctx, teacherID) // []db.ClassesOfTeacherRow, có kiểu
c, err := q.CreateClass(ctx, db.CreateClassParams{TeacherID: 7, Name: "5A", Capacity: 35})

sqlc kiểm tra câu SQL với lược đồ ngay lúc sinh code: gõ sai tên cột là báo lỗi trước khi chạy.

Còn ORM thì sao?

Go có ORM (GORM, ent, Bun). GORM rất phổ biến, nhất là ở các dự án CRUD. Nhưng nhiều đội Go ưa SQL thuần + sqlc / pgx vì thấy rõ câu SQL nào được chạy, dễ tối ưu, và không bị ORM sinh câu truy vấn bất ngờ. Ở công ty, bạn sẽ dùng thứ đội đang dùng. Hiểu SQL thì dùng cái nào cũng được.

- Cả hai tạo được tài khoản + Cả hai SELECT đều thấy "chưa có", một INSERT thành công, cái còn lại bị UNIQUE từ chối với lỗi 23505, nên code phải xử lý lỗi đó thành 409 - Request thứ hai tự đợi request đầu xong rồi mới SELECT ? Kiểm tra rồi mới làm không an toàn khi chạy song song. Ràng buộc UNIQUE mới là người gác cổng thật. Có thể bỏ luôn bước SELECT và chỉ xử lý lỗi 23505.

Bài tập

Repository cho bài tập

Với lược đồ và dữ liệu mẫu bài 14, viết AssignmentStore dùng pgx, có ba method:

  • Create(ctx, teacherID, classID, title, due): giao bài mới. Lớp không thuộc giáo viên đó thì trả ErrNotFound.
  • ListByClass(ctx, teacherID, classID, page, size): trả về một trang bài tập (mới nhất trước) và tổng số bài.
  • Delete(ctx, teacherID, id): xoá bài. Bài không thuộc lớp của giáo viên đó thì trả ErrNotFound.

Mọi kiểm tra "lớp có thuộc giáo viên này không" phải nằm ngay trong câu SQL, không SELECT kiểm tra trước. Viết main thử cả trường hợp Thầy Nam (id 8) động vào lớp 42 của Cô Hoa (id 7).

Gợi ý từng bước

  1. Tạo dự án như phần "Chạy thử" ở trên, hai file: store.go (kiểu AssignmentStore chứa *pgxpool.Pool) và main.go.
  2. Create: dùng INSERT ... SELECT thay vì INSERT ... VALUES. Phần SELECT c.id, $3, $4 FROM classes c WHERE c.id = $1 AND c.teacher_id = $2 chỉ ra một dòng khi lớp đúng là của giáo viên, nên sai giáo viên thì không có gì được thêm, và RETURNING không trả dòng nào (pgx.ErrNoRows).
  3. ListByClass: hai câu. Một câu count(*) lấy tổng, một câu lấy trang với LIMIT size OFFSET (page-1)*size. Cả hai JOIN classes để lọc teacher_id.
  4. Delete: Postgres có cú pháp DELETE FROM a USING c WHERE ... để xoá có điều kiện trên bảng khác. Đọc RowsAffected(): 0 là không có gì bị xoá.
Xem lời giải 
package main

import (
    "context"
    "errors"
    "time"

    "github.com/jackc/pgx/v5"
    "github.com/jackc/pgx/v5/pgxpool"
)

var ErrNotFound = errors.New("không tìm thấy")

type Assignment struct {
    ID        int64
    ClassID   int64
    Title     string
    DueDate   time.Time
    CreatedAt time.Time
}

type AssignmentStore struct {
    db *pgxpool.Pool
}

// Create chỉ thêm được bài vào lớp của chính giáo viên đó.
// INSERT ... SELECT: nếu lớp không thuộc giáo viên, SELECT không ra dòng nào, nên không thêm gì
func (s *AssignmentStore) Create(ctx context.Context, teacherID, classID int64, title string, due time.Time) (Assignment, error) {
    a := Assignment{ClassID: classID, Title: title, DueDate: due}
    err := s.db.QueryRow(ctx, `
        INSERT INTO assignments (class_id, title, due_date)
        SELECT c.id, $3, $4 FROM classes c
        WHERE c.id = $1 AND c.teacher_id = $2
        RETURNING id, created_at`,
        classID, teacherID, title, due,
    ).Scan(&a.ID, &a.CreatedAt)
    if errors.Is(err, pgx.ErrNoRows) {
        return Assignment{}, ErrNotFound // lớp không có, hoặc không phải lớp của giáo viên này
    }
    return a, err
}

// ListByClass trả về một trang bài tập và tổng số bài
func (s *AssignmentStore) ListByClass(ctx context.Context, teacherID, classID int64, page, size int) ([]Assignment, int, error) {
    var total int
    err := s.db.QueryRow(ctx, `
        SELECT count(*) FROM assignments a
        JOIN classes c ON c.id = a.class_id
        WHERE a.class_id = $1 AND c.teacher_id = $2`, classID, teacherID).Scan(&total)
    if err != nil {
        return nil, 0, err
    }
    rows, err := s.db.Query(ctx, `
        SELECT a.id, a.class_id, a.title, a.due_date, a.created_at
        FROM assignments a
        JOIN classes c ON c.id = a.class_id
        WHERE a.class_id = $1 AND c.teacher_id = $2
        ORDER BY a.due_date DESC, a.id DESC
        LIMIT $3 OFFSET $4`, classID, teacherID, size, (page-1)*size)
    if err != nil {
        return nil, 0, err
    }
    // CollectRows đọc hết các dòng và tự rows.Close(). RowToStructByPos: cột thứ n vào trường thứ n
    items, err := pgx.CollectRows(rows, pgx.RowToStructByPos[Assignment])
    return items, total, err
}

// Delete chỉ xoá được bài của lớp mình dạy
func (s *AssignmentStore) Delete(ctx context.Context, teacherID, id int64) error {
    tag, err := s.db.Exec(ctx, `
        DELETE FROM assignments a USING classes c
        WHERE a.id = $1 AND c.id = a.class_id AND c.teacher_id = $2`, id, teacherID)
    if err != nil {
        return err
    }
    if tag.RowsAffected() == 0 {
        return ErrNotFound
    }
    return nil
}

    ctx := context.Background()
    pool, err := pgxpool.New(ctx, os.Getenv("DATABASE_URL"))
    if err != nil {
        fmt.Println(err)
        os.Exit(1)
    }
    defer pool.Close()
    store := &AssignmentStore{db: pool}

    // Cô Hoa (7) giao bài mới cho lớp 42 của mình
    due := time.Date(2026, 11, 10, 0, 0, 0, 0, time.UTC)
    a, err := store.Create(ctx, 7, 42, "Toán trang 20", due)
    fmt.Println("tạo:", a.ID, a.Title, err)

    // Thầy Nam (8) cố giao bài vào lớp 42 của Cô Hoa
    _, err = store.Create(ctx, 8, 42, "Bài lạ", due)
    fmt.Println("thầy Nam tạo vào lớp 42:", err)

    // Trang 1, mỗi trang 2 bài
    items, total, err := store.ListByClass(ctx, 7, 42, 1, 2)
    fmt.Println("tổng:", total, err)
    for _, it := range items {
        fmt.Println(" -", it.ID, it.Title, it.DueDate.Format("02/01/2006"))
    }
    items, _, _ = store.ListByClass(ctx, 7, 42, 2, 2)
    fmt.Println("trang 2:", len(items), "bài, bài đầu:", items[0].Title)

    // Thầy Nam nhìn lớp 42: không thấy gì, như thể lớp không tồn tại
    _, total, _ = store.ListByClass(ctx, 8, 42, 1, 20)
    fmt.Println("thầy Nam xem lớp 42:", total, "bài")

    fmt.Println("thầy Nam xoá bài 9:", store.Delete(ctx, 8, 9))
    fmt.Println("cô Hoa xoá bài 9:", store.Delete(ctx, 7, 9))
    fmt.Println("cô Hoa xoá lại bài 9:", store.Delete(ctx, 7, 9))
}
