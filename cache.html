Cache và Redis

Khi nào cần cache, cache-aside, TTL, xoá cache, cache stampede, giới hạn tần suất bằng Redis.

Học xong bài này bạn sẽ:
  • Áp dụng cache-aside với Redis
  • Biết các cách cache làm sai dữ liệu
  • Chặn hàng nghìn request trùng bằng singleflight

#Khi nào cần cache

Cache là giữ sẵn kết quả của một việc tốn kém để lần sau trả ngay. Nhưng cache cũng là nguồn gốc của rất nhiều lỗi "dữ liệu cũ". Câu đùa nổi tiếng của Phil Karlton: "Chỉ có hai việc khó trong khoa học máy tính: xoá cache cho đúng và đặt tên."

Chỉ cache khi đo được là cần:

  • Dữ liệu được đọc nhiều hơn ghi rất nhiều (cấu hình, danh mục, trang chủ, hồ sơ công khai).
  • Tính toán đắt (báo cáo tổng hợp, gọi API bên ngoài có tính phí hoặc chậm).
  • Chấp nhận được việc dữ liệu cũ đi vài giây / vài phút.

Trước khi thêm cache, hãy thử thêm index hoặc sửa N+1 (bài 19). Một truy vấn 0,1ms có index thì không cần cache. Cache che đi một truy vấn chậm, nhưng không sửa được nó.

#Các tầng cache trong một hệ thống

Bạn đã quen với tầng đầu: Cache-Control: max-age=31536000, immutable cho file JS có hash trong tên, và React Query / SWR giữ dữ liệu API trong bộ nhớ của tab. Backend thêm các tầng phía sau.

#Redis

Redis là một kho key-value chạy trong RAM, mỗi lệnh thường mất dưới 1ms. Ngoài làm cache, nó còn được dùng cho: phiên đăng nhập, đếm giới hạn tần suất, khoá phân tán, hàng đợi đơn giản, bảng xếp hạng.

#Cache-aside: mẫu dùng nhiều nhất

Go · trích đoạn
func (s *Service) ClassSummary(ctx context.Context, classID int64) (Summary, error) {
    key := fmt.Sprintf("class:%d:summary:v1", classID)

    if b, err := s.rdb.Get(ctx, key).Bytes(); err == nil {
        var sum Summary
        if json.Unmarshal(b, &sum) == nil {
            return sum, nil // hit
        }
    } else if !errors.Is(err, redis.Nil) {
        // Redis lỗi: ghi log rồi đi tiếp vào database, KHÔNG làm hỏng request
        slog.WarnContext(ctx, "redis get", "err", err)
    }

    sum, err := s.store.ComputeSummary(ctx, classID)
    if err != nil {
        return Summary{}, err
    }
    if b, err := json.Marshal(sum); err == nil {
        s.rdb.Set(ctx, key, b, 5*time.Minute) // lỗi ghi cache cũng không làm hỏng request
    }
    return sum, nil
}

Chạy thử. Chương trình hoàn chỉnh dưới giả lập một truy vấn database mất 300ms và đo thời gian khi có / không có cache:

Terminal
mkdir cache-demo && cd cache-demo
go mod init cache-demo
go get github.com/redis/go-redis/v9
main.go
package main

import (
    "context"
    "encoding/json"
    "errors"
    "fmt"
    "os"
    "time"

    "github.com/redis/go-redis/v9"
)

type Summary struct {
    ClassID   int64   `json:"classId"`
    Students  int     `json:"students"`
    AvgScore  float64 `json:"avgScore"`
}

// Giả lập truy vấn tổng hợp nặng trên database: mất 300ms
func computeSummary(ctx context.Context, classID int64) (Summary, error) {
    time.Sleep(300 * time.Millisecond)
    return Summary{ClassID: classID, Students: 4, AvgScore: 6.4}, nil
}

// Cache-aside: thử cache, không có thì tính rồi ghi vào cache
func classSummary(ctx context.Context, rdb *redis.Client, classID int64) (Summary, string, error) {
    key := fmt.Sprintf("class:%d:summary:v1", classID)

    b, err := rdb.Get(ctx, key).Bytes()
    if err == nil {
        var sum Summary
        if json.Unmarshal(b, &sum) == nil {
            return sum, "hit", nil // có trong cache
        }
    } else if !errors.Is(err, redis.Nil) { // redis.Nil = khoá không tồn tại, không phải lỗi thật
        fmt.Println("  (redis lỗi, đọc thẳng database:", err, ")")
    }

    sum, err := computeSummary(ctx, classID) // miss: hỏi database
    if err != nil {
        return Summary{}, "", err
    }
    if b, err := json.Marshal(sum); err == nil {
        rdb.Set(ctx, key, b, 5*time.Minute) // ghi cache, hết hạn sau 5 phút
    }
    return sum, "miss", nil
}

func main() {
    ctx := context.Background()
    rdb := redis.NewClient(&redis.Options{Addr: os.Getenv("REDIS_ADDR")})
    defer rdb.Close()
    rdb.Del(ctx, "class:42:summary:v1") // bắt đầu sạch

    for i := 1; i <= 3; i++ {
        start := time.Now()
        sum, kind, _ := classSummary(ctx, rdb, 42)
        fmt.Printf("lần %d: %s, %+v, mất %v\n", i, kind, sum, time.Since(start).Round(time.Millisecond))
    }

    rdb.Del(ctx, "class:42:summary:v1") // dữ liệu đổi (có bài nộp mới): xoá cache
    start := time.Now()
    _, kind, _ := classSummary(ctx, rdb, 42)
    fmt.Printf("sau khi xoá cache: %s, mất %v\n", kind, time.Since(start).Round(time.Millisecond))

    ttl := rdb.TTL(ctx, "class:42:summary:v1").Val()
    fmt.Println("cache còn sống:", ttl.Round(time.Second))
}

lần 2: hit, {ClassID:42 Students:4 AvgScore:6.4}, mất 1ms

Lần đầu phải hỏi database (300ms), hai lần sau lấy từ Redis (dưới 1ms, in ra 0s vì làm tròn). Xoá cache thì lần sau lại phải hỏi database. Xem tận mắt dữ liệu trong Redis: mở redis-cli và gõ GET class:42:summary:v1, sẽ thấy chuỗi JSON.

Cache phải là thứ có cũng được, mất cũng được. Redis sập thì hệ thống chậm đi, không được chết theo. Và luôn có TTL: dù bạn quên xoá cache ở đâu đó, dữ liệu sai cũng chỉ tồn tại tối đa bằng TTL.

Hậu tố :v1 trong khoá: khi đổi cấu trúc Summary, tăng lên v2, bản code mới không đọc nhầm dữ liệu cache dạng cũ.

#Xoá cache khi dữ liệu đổi

Go · trích đoạn
func (s *Service) Submit(ctx context.Context, sub Submission) error {
    if err := s.store.Submit(ctx, sub); err != nil { // 1. ghi database trước
        return err
    }
    s.rdb.Del(ctx, fmt.Sprintf("class:%d:summary:v1", sub.ClassID)) // 2. rồi xoá cache
    return nil
}

Vì sao xoá thay vì ghi giá trị mới vào cache? Vì hai request cập nhật cùng lúc có thể ghi cache theo thứ tự ngược với database, để lại giá trị cũ trong cache. Xoá thì lần đọc sau tự lấy giá trị đúng từ database.

Vẫn còn một khe hở nhỏ: một request đọc lấy giá trị cũ từ database, request khác ghi và xoá cache, rồi request đọc kia mới ghi giá trị cũ vào cache. TTL giới hạn thời gian sai. Với dữ liệu tuyệt đối không được sai (số dư, tồn kho, quyền truy cập), đừng cache, hoặc chỉ dùng cache để hiển thị, còn quyết định thì luôn đọc từ database.

#Cache stampede: hàng nghìn request cùng tính lại

Trang chủ được cache 60 giây, 2.000 request mỗi giây. Đúng lúc cache hết hạn, 2.000 request cùng thấy "miss" và cùng chạy truy vấn nặng. Database quá tải, chậm lại, nhiều request hơn bị miss... Đây gọi là cache stampede (hay thundering herd).

Cách chặn trong một tiến trình: singleflight. Các lời gọi cùng khoá đang chạy thì chờ và dùng chung kết quả của lời gọi đầu tiên.

Chạy thử để thấy hiệu quả (không cần Redis, cần go get golang.org/x/sync):

main.go
package main

import (
    "fmt"
    "sync"
    "sync/atomic"
    "time"

    "golang.org/x/sync/singleflight"
)

var dbCalls atomic.Int64 // đếm số lần thật sự hỏi database

func loadHomeFeed() (string, error) {
    dbCalls.Add(1)
    time.Sleep(200 * time.Millisecond) // truy vấn nặng
    return "bảng tin trang chủ", nil
}

func main() {
    var group singleflight.Group
    var wg sync.WaitGroup

    // 1.000 request cùng lúc, đúng lúc cache vừa hết hạn
    for range 1000 {
        wg.Add(1)
        go func() {
            defer wg.Done()
            // Cùng khoá "home-feed": chỉ MỘT goroutine chạy loadHomeFeed, số còn lại chờ và nhận chung kết quả
            v, err, _ := group.Do("home-feed", func() (any, error) {
                return loadHomeFeed()
            })
            _ = v.(string) // Do trả về any, đổi lại thành string
            _ = err
        }()
    }
    wg.Wait()
    fmt.Println("1000 request, số lần hỏi database:", dbCalls.Load())

    // So sánh: không có singleflight
    dbCalls.Store(0)
    for range 1000 {
        wg.Add(1)
        go func() {
            defer wg.Done()
            loadHomeFeed()
        }()
    }
    wg.Wait()
    fmt.Println("không có singleflight, số lần hỏi database:", dbCalls.Load())
}
Terminal — kết quả khi chạy
1000 request, số lần hỏi database: 1
không có singleflight, số lần hỏi database: 1000

Cách dùng trong service thật (trích đoạn):

Go · trích đoạn
import "golang.org/x/sync/singleflight"

type Service struct {
    group singleflight.Group
    // ...
}

func (s *Service) HomeFeed(ctx context.Context) (Feed, error) {
    v, err, _ := s.group.Do("home-feed", func() (any, error) {
        return s.loadHomeFeed(ctx) // chỉ MỘT goroutine chạy, những cái khác chờ kết quả này
    })
    if err != nil {
        return Feed{}, err
    }
    return v.(Feed), nil
}

Các cách khác: thêm ngẫu nhiên vào TTL (5 phút ± 30 giây) để các khoá không hết hạn cùng lúc, và làm mới cache ở nền trước khi hết hạn.

Package singleflight vốn nằm trong mã nguồn của groupcache, thư viện cache do Brad Fitzpatrick viết khi ở Google, dùng cho dl.google.com. Nó gom các yêu cầu trùng nhau để chỉ một yêu cầu thật sự đi tới nguồn.

#Cache trong bộ nhớ của server

Với dữ liệu nhỏ, rất nóng, ít đổi (cấu hình, danh sách môn học mặc định), map trong bộ nhớ có mutex nhanh hơn Redis nhiều (không qua mạng). Nhược điểm: mỗi bản server có bản riêng, không xoá đồng loạt được. Chỉ dùng với TTL ngắn và dữ liệu chấp nhận lệch giữa các máy.

#Giới hạn tần suất (rate limit)

Chặn dò mật khẩu, spam gửi OTP / email, một người dùng ngốn hết tài nguyên. Hai cách:

Trong một tiến trình: token bucket

Go · trích đoạn
import "golang.org/x/time/rate"

// Mỗi giây thêm 5 "vé", tối đa dồn 10 vé: cho phép bùng 10 request rồi 5 request mỗi giây
lim := rate.NewLimiter(5, 10)
if !lim.Allow() {
    w.Header().Set("Retry-After", "1")
    http.Error(w, "Quá nhiều yêu cầu", http.StatusTooManyRequests)
    return
}

rate.NewLimiter(5, 10): hình dung một cái xô chứa tối đa 10 vé, mỗi giây được bỏ thêm 5 vé. Mỗi request lấy một vé, hết vé thì bị từ chối. Nhờ vậy cho phép bùng lên một chút (10 request liền) nhưng về lâu dài không quá 5 request mỗi giây.

Dùng chung nhiều server: đếm trong Redis

Go · trích đoạn
// Cửa sổ cố định: tối đa 5 lần đăng nhập sai mỗi 15 phút cho mỗi email
func (s *Service) allowLogin(ctx context.Context, email string) (bool, error) {
    key := "rl:login:" + email
    n, err := s.rdb.Incr(ctx, key).Result() // INCR là nguyên tử
    if err != nil {
        return true, err // Redis lỗi: tuỳ chính sách, ở đây cho qua và ghi log
    }
    if n == 1 {
        s.rdb.Expire(ctx, key, 15*time.Minute) // lần đầu trong cửa sổ thì đặt hạn
    }
    return n <= 5, nil
}

Chạy thử cả hai cách (cần Redis, cài thêm go get golang.org/x/time):

main.go
package main

import (
    "context"
    "fmt"
    "os"
    "time"

    "github.com/redis/go-redis/v9"
    "golang.org/x/time/rate"
)

// Cửa sổ cố định trong Redis: tối đa 5 lần đăng nhập mỗi 15 phút cho mỗi email
func allowLogin(ctx context.Context, rdb *redis.Client, email string) (bool, error) {
    key := "rl:login:" + email
    n, err := rdb.Incr(ctx, key).Result() // tăng 1, trả về giá trị mới. Nguyên tử: không ai chen vào được
    if err != nil {
        return true, err
    }
    if n == 1 {
        rdb.Expire(ctx, key, 15*time.Minute) // lần đầu trong cửa sổ: đặt hạn
    }
    return n <= 5, nil
}

func main() {
    // 1. Token bucket trong một tiến trình: thêm 5 vé mỗi giây, dồn tối đa 10 vé
    lim := rate.NewLimiter(5, 10)
    allowed := 0
    for range 15 { // 15 request dồn dập trong cùng một lúc
        if lim.Allow() {
            allowed++
        }
    }
    fmt.Println("token bucket: 15 request dồn dập, cho qua", allowed)
    time.Sleep(time.Second) // chờ 1 giây: thêm 5 vé
    allowed = 0
    for range 15 {
        if lim.Allow() {
            allowed++
        }
    }
    fmt.Println("token bucket: sau 1 giây, cho qua thêm", allowed)

    // 2. Đếm trong Redis: dùng chung cho mọi bản server
    ctx := context.Background()
    rdb := redis.NewClient(&redis.Options{Addr: os.Getenv("REDIS_ADDR")})
    defer rdb.Close()
    rdb.Del(ctx, "rl:login:lan@gmail.com")
    for i := 1; i <= 7; i++ {
        ok, _ := allowLogin(ctx, rdb, "lan@gmail.com")
        fmt.Printf("đăng nhập lần %d: cho phép = %v\n", i, ok)
    }
}

Token bucket nằm trong bộ nhớ một tiến trình: chạy 3 bản server thì mỗi bản đếm riêng, người dùng được gấp 3. Bộ đếm Redis thì mọi bản server cùng nhìn một con số. Chạy lại chương trình trong vòng 15 phút: dòng rdb.Del ở đầu xoá bộ đếm nên lại được 5 lần. Bỏ dòng đó đi thì từ lần chạy thứ hai mọi lần đều bị chặn.

Đoạn trên có điểm yếu nhỏ: nếu server chết giữa INCR và EXPIRE, khoá không bao giờ hết hạn và người đó bị khoá mãi. Cách chắc chắn là gộp hai lệnh vào một script Lua hoặc một MULTI/EXEC pipeline để Redis chạy nguyên khối. Đây chính là loại lỗi "hai bước rời nhau" bạn đã gặp ở database.

Các dịch vụ lớn trả về header cho client biết giới hạn: GitHub API dùng X-RateLimit-Limit, X-RateLimit-Remaining, X-RateLimit-Reset. Có một bản nháp chuẩn IETF cho header RateLimit đang được nhiều nơi áp dụng.

#Cache HTTP cho client

Go · trích đoạn
// Dữ liệu công khai, ít đổi: cho CDN và trình duyệt cache 5 phút
w.Header().Set("Cache-Control", "public, max-age=300")

// Dữ liệu riêng của người dùng: CDN không được cache
w.Header().Set("Cache-Control", "private, no-store")

Một lỗi nghiêm trọng có thật: CDN cache nhầm response chứa dữ liệu cá nhân (vì thiếu private / no-store), và người dùng khác truy cập cùng URL nhận được trang của người khác. Mọi response cần đăng nhập nên mặc định Cache-Control: no-store.

Câu hỏi nhanh. Redis bị sập 5 phút. Với mẫu cache-aside viết đúng, chuyện gì xảy ra?

Cache lỗi thì bỏ qua và đọc nguồn sự thật. Đó là lý do cần theo dõi tải database và có rate limit, phòng khi cache biến mất đột ngột.

#Bài tập

Cache có singleflight và TTL lệch ngẫu nhiên

Gộp ba ý trong bài thành một hàm dùng lại được: GetOrLoad[T any](ctx, rdb, group, key, ttl, load) (T, error).

  • thử đọc từ Redis trước.
  • miss thì dùng singleflight gọi load: 100 request cùng lúc chỉ gọi load một lần.
  • ghi kết quả vào Redis với TTL cộng thêm một khoảng ngẫu nhiên tối đa 10%, để các khoá không cùng hết hạn một lúc.
  • Redis lỗi thì vẫn trả được kết quả từ load.

Gợi ý từng bước

  1. Hàm generic (bài 6) vì mỗi chỗ dùng cache một kiểu dữ liệu khác: Feed, Summary... load có kiểu func(context.Context) (T, error).
  2. Bước đọc giống classSummary ở trên: rdb.Get(ctx, key).Bytes() rồi json.Unmarshal vào một biến kiểu T.
  3. g.Do(key, func() (any, error) {...}) trả về any, đổi lại thành T bằng v.(T).
  4. Số ngẫu nhiên: rand.Int64N(n) của package math/rand/v2 cho số từ 0 đến n-1. time.Duration bản chất là số nano giây (int64), nên đổi qua lại được.
  5. Viết main kiểm tra: 100 goroutine cùng gọi, đếm số lần load chạy. Rồi thử với một Redis không tồn tại (Addr: "localhost:1").

cache.go
package main

import (
    "context"
    "encoding/json"
    "math/rand/v2"
    "time"

    "github.com/redis/go-redis/v9"
    "golang.org/x/sync/singleflight"
)

// GetOrLoad: thử Redis. Miss thì chỉ MỘT goroutine gọi load (singleflight),
// ghi kết quả vào Redis với TTL lệch ngẫu nhiên tối đa 10%. Redis lỗi vẫn trả được kết quả.
func GetOrLoad[T any](ctx context.Context, rdb *redis.Client, g *singleflight.Group,
    key string, ttl time.Duration, load func(context.Context) (T, error)) (T, error) {

    var zero T

    // 1. Thử cache. Lỗi gì (không có khoá, Redis sập, JSON hỏng) cũng bỏ qua, đi tiếp bước 2
    if b, err := rdb.Get(ctx, key).Bytes(); err == nil {
        var v T
        if json.Unmarshal(b, &v) == nil {
            return v, nil
        }
    }

    // 2. Miss: các goroutine cùng khoá chờ chung một lần load
    v, err, _ := g.Do(key, func() (any, error) {
        val, err := load(ctx)
        if err != nil {
            return nil, err
        }
        if b, err := json.Marshal(val); err == nil {
            jitter := time.Duration(rand.Int64N(int64(ttl / 10))) // 0 đến 10% của ttl
            rdb.Set(ctx, key, b, ttl+jitter)                      // lỗi ghi cache: bỏ qua
        }
        return val, nil
    })
    if err != nil {
        return zero, err
    }
    return v.(T), nil
}
main.go
package main

import (
    "context"
    "fmt"
    "os"
    "sync"
    "sync/atomic"
    "time"

    "github.com/redis/go-redis/v9"
    "golang.org/x/sync/singleflight"
)

type Feed struct {
    Items []string `json:"items"`
}

func main() {
    ctx := context.Background()
    rdb := redis.NewClient(&redis.Options{Addr: os.Getenv("REDIS_ADDR")})
    defer rdb.Close()
    rdb.Del(ctx, "home-feed:v1")
    var g singleflight.Group

    var loads atomic.Int64
    load := func(ctx context.Context) (Feed, error) { // giả lập database chậm
        loads.Add(1)
        time.Sleep(200 * time.Millisecond)
        return Feed{Items: []string{"Bài Toán mới", "Lịch họp phụ huynh"}}, nil
    }

    // 1. 100 request cùng lúc khi cache trống
    var wg sync.WaitGroup
    for range 100 {
        wg.Add(1)
        go func() {
            defer wg.Done()
            if _, err := GetOrLoad(ctx, rdb, &g, "home-feed:v1", time.Minute, load); err != nil {
                fmt.Println("lỗi:", err)
            }
        }()
    }
    wg.Wait()
    fmt.Println("100 request đồng thời, số lần load:", loads.Load())

    // 2. Request sau: lấy từ Redis, không load nữa
    feed, _ := GetOrLoad(ctx, rdb, &g, "home-feed:v1", time.Minute, load)
    fmt.Println("request sau:", feed.Items, "| số lần load:", loads.Load())
    fmt.Println("TTL trong Redis:", rdb.TTL(ctx, "home-feed:v1").Val().Round(time.Second), "(từ 60 đến 66 giây)")

    // 3. Redis sập: vẫn trả được kết quả, chỉ chậm hơn
    dead := redis.NewClient(&redis.Options{Addr: "localhost:1", MaxRetries: -1})
    feed, err := GetOrLoad(ctx, dead, &g, "home-feed:v1", time.Minute, load)
    fmt.Println("Redis sập:", feed.Items, err)
}

Giải thích: hai dòng redis: ... connection refused là thư viện go-redis tự ghi log khi không kết nối được (một lần lúc đọc, một lần lúc ghi). Dù vậy, dòng cuối vẫn có dữ liệu và lỗi là <nil>: cache sập, người dùng vẫn được phục vụ, chỉ chậm hơn. TTL mỗi lần chạy một khác, trong khoảng 60 đến 66 giây.

Lưu ý: hàm load chạy với ctx của request đầu tiên. Nếu request đó bị huỷ, các request đang chờ cũng nhận lỗi theo. Ở hệ thống thật người ta dùng context.WithoutCancel(ctx) (Go 1.21+) kèm timeout riêng cho việc tải chung.
