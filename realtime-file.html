WebSocket, upload file và gửi email

WebSocket / SSE cho realtime, hub nhiều client, mở rộng nhiều server bằng Redis pub/sub; upload file trực tiếp lên S3 bằng presigned URL; gửi email qua hàng đợi.

Học xong bài này bạn sẽ:
  • Đẩy dữ liệu realtime xuống trình duyệt
  • Upload file lớn không qua server
  • Gửi email tin cậy

#Ba việc mọi sản phẩm đều cần

Bài này gom ba việc "nhìn thì đơn giản" nhưng làm sai rất tốn kém:

  • Đẩy dữ liệu realtime xuống trình duyệt: giáo viên thấy bài nộp mới ngay, không cần tải lại trang.
  • Upload file: phụ huynh nộp ảnh bài làm, giáo viên đính kèm PDF.
  • Gửi email: xác nhận, biên lai, nhắc việc.

Phần lớn nhu cầu "realtime" chỉ là server báo cho client. SSE đơn giản hơn WebSocket nhiều: dùng HTTP bình thường, cookie và middleware xác thực vẫn chạy, trình duyệt tự kết nối lại. Chỉ chọn WebSocket khi client cần gửi liên tục.

SSE bằng thư viện chuẩn

main.go
package main

import (
    "fmt"
    "log"
    "net/http"
    "time"
)

func events(w http.ResponseWriter, r *http.Request) {
    w.Header().Set("Content-Type", "text/event-stream")
    w.Header().Set("Cache-Control", "no-cache")
    w.Header().Set("X-Accel-Buffering", "no") // báo Nginx đừng gom response lại

    rc := http.NewResponseController(w) // Go 1.20+: Flush, đặt deadline ghi
    rc.SetWriteDeadline(time.Time{})     // bỏ WriteTimeout của server cho kết nối dài này

    tick := time.NewTicker(time.Second)
    defer tick.Stop()
    for i := 1; ; i++ {
        select {
        case <-r.Context().Done(): // trình duyệt đóng tab
            log.Println("client ngắt")
            return
        case t := <-tick.C:
            // Định dạng SSE: "id:", "event:", "data:", kết thúc bằng một dòng trống
            fmt.Fprintf(w, "id: %d\nevent: tick\ndata: {\"n\":%d,\"at\":%q}\n\n", i, i, t.Format("15:04:05"))
            if err := rc.Flush(); err != nil { // đẩy ngay xuống client, không chờ đầy bộ đệm
                return
            }
        }
    }
}

func main() {
    http.HandleFunc("GET /events", events)
    log.Fatal(http.ListenAndServe(":8080", nil))
}

WebSocket: hub nhiều client

Thư viện: github.com/coder/websocket (trước là nhooyr.io/websocket, gọn, hỗ trợ context) hoặc github.com/gorilla/websocket (lâu đời, rất nhiều code cũ dùng). Bài này dùng coder/websocket.

ws.go
package main

import (
    "context"
    "encoding/json"
    "log/slog"
    "net/http"
    "sync"
    "time"

    "github.com/coder/websocket"
)

type client struct {
    send chan []byte // bộ đệm ra: hub ghi vào, goroutine ghi của client đọc ra
}

type Hub struct {
    mu    sync.Mutex
    rooms map[string]map[*client]struct{}
}

func NewHub() *Hub { return &Hub{rooms: map[string]map[*client]struct{}{}} }

func (h *Hub) join(room string, c *client) {
    h.mu.Lock()
    defer h.mu.Unlock()
    if h.rooms[room] == nil {
        h.rooms[room] = map[*client]struct{}{}
    }
    h.rooms[room][c] = struct{}{}
}

func (h *Hub) leave(room string, c *client) {
    h.mu.Lock()
    defer h.mu.Unlock()
    delete(h.rooms[room], c)
    if len(h.rooms[room]) == 0 {
        delete(h.rooms, room)
    }
}

func (h *Hub) Publish(room string, msg any) {
    b, _ := json.Marshal(msg)
    h.mu.Lock()
    defer h.mu.Unlock()
    for c := range h.rooms[room] {
        select {
        case c.send <- b:
        default:
            // client quá chậm (mạng yếu), bộ đệm đầy: bỏ tin thay vì làm treo cả hub
            slog.Warn("bỏ tin cho client chậm", "room", room)
        }
    }
}

func (h *Hub) Handle(w http.ResponseWriter, r *http.Request) {
    room := "class:" + r.PathValue("id")
    // QUAN TRỌNG: kiểm tra quyền vào phòng TRƯỚC khi nâng cấp kết nối (như mọi API khác)

    conn, err := websocket.Accept(w, r, &websocket.AcceptOptions{
        OriginPatterns: []string{"app.sobaitap.vn", "localhost:5173"}, // chống trang lạ mở WebSocket bằng cookie của người dùng
    })
    if err != nil {
        return
    }
    defer conn.CloseNow()

    c := &client{send: make(chan []byte, 32)}
    h.join(room, c)
    defer h.leave(room, c)

    ctx, cancel := context.WithCancel(r.Context())
    defer cancel()

    // Goroutine đọc: bắt buộc phải đọc để xử lý ping/pong và phát hiện client đóng
    go func() {
        defer cancel()
        for {
            if _, _, err := conn.Read(ctx); err != nil {
                return
            }
            // ứng dụng này không nhận tin từ client, bỏ qua
        }
    }()

    ping := time.NewTicker(25 * time.Second) // giữ kết nối sống qua proxy (nhiều proxy cắt sau 60 giây im lặng)
    defer ping.Stop()
    for {
        select {
        case b := <-c.send:
            wctx, wcancel := context.WithTimeout(ctx, 5*time.Second)
            err := conn.Write(wctx, websocket.MessageText, b)
            wcancel()
            if err != nil {
                return
            }
        case <-ping.C:
            if err := conn.Ping(ctx); err != nil {
                return
            }
        case <-ctx.Done():
            conn.Close(websocket.StatusNormalClosure, "")
            return
        }
    }
}

Ba lỗi bảo mật hay gặp với WebSocket: (1) không kiểm tra quyền vào phòng, ai biết id lớp cũng nghe được. (2) Không kiểm tra Origin: trang độc hại mở WebSocket tới API của bạn, trình duyệt tự gửi kèm cookie (giống CSRF). (3) Tin gửi xuống có dữ liệu của người khác: chỉ gửi đúng thứ người nghe được phép xem.

Chạy nhiều server: Redis pub/sub

Hub ở trên nằm trong bộ nhớ một server. Có 3 bản server sau load balancer thì giáo viên nối vào server A, phụ huynh nộp bài ở server B: hub của B không biết giáo viên. Cách sửa: mọi server cùng nghe một kênh chung.

Go · trích đoạn
// Server nào nhận việc cũng chỉ publish lên Redis, KHÔNG gọi hub trực tiếp
func (s *Service) notify(ctx context.Context, room string, msg any) {
    b, _ := json.Marshal(msg)
    s.rdb.Publish(ctx, "ws:"+room, b)
}

// Mỗi server chạy một goroutine nghe mọi kênh ws:* và đưa vào hub của mình
func (h *Hub) RunRedis(ctx context.Context, rdb *redis.Client) {
    sub := rdb.PSubscribe(ctx, "ws:*")
    defer sub.Close()
    for m := range sub.Channel() { // channel đóng khi sub.Close()
        room := strings.TrimPrefix(m.Channel, "ws:")
        h.publishRaw(room, []byte(m.Payload))
    }
}

Redis pub/sub không lưu tin: client đang mất kết nối thì mất tin. Vì vậy client khi nối lại nên tải lại trạng thái từ API (số bài chờ chấm), chỉ dùng tin realtime để cập nhật nhanh. Muốn không mất tin thì dùng Redis Streams hoặc NATS JetStream (bài 24).

#Upload file đúng cách: presigned URL

Cách đơn giản nhất là trình duyệt gửi file lên server Go, server Go lưu file. Nhưng với ảnh 5MB, hàng nghìn phụ huynh nộp cùng tối, server tốn băng thông, RAM, và giữ kết nối rất lâu với mạng điện thoại chậm. Cách mọi hệ thống lớn dùng: trình duyệt upload thẳng lên kho file (S3, Cloudflare R2, Google Cloud Storage, MinIO) bằng một URL có chữ ký do server cấp.

Terminal
go get github.com/aws/aws-sdk-go-v2/config github.com/aws/aws-sdk-go-v2/service/s3
upload.go
package main

import (
    "context"
    "fmt"
    "net/http"
    "time"

    "github.com/aws/aws-sdk-go-v2/aws"
    "github.com/aws/aws-sdk-go-v2/config"
    "github.com/aws/aws-sdk-go-v2/service/s3"
    "github.com/oklog/ulid/v2"
)

const maxSize = 5 << 20 // 5 MB

var allowed = map[string]string{"image/jpeg": ".jpg", "image/png": ".png", "image/webp": ".webp", "application/pdf": ".pdf"}

type Uploader struct {
    s3     *s3.Client
    presig *s3.PresignClient
    bucket string
}

func NewUploader(ctx context.Context, bucket, endpoint string) (*Uploader, error) {
    cfg, err := config.LoadDefaultConfig(ctx) // đọc AWS_ACCESS_KEY_ID, AWS_SECRET_ACCESS_KEY, AWS_REGION
    if err != nil {
        return nil, err
    }
    c := s3.NewFromConfig(cfg, func(o *s3.Options) {
        if endpoint != "" { // R2 / MinIO: đổi địa chỉ, giao thức vẫn là S3
            o.BaseEndpoint = aws.String(endpoint)
            o.UsePathStyle = true
        }
    })
    return &Uploader{s3: c, presig: s3.NewPresignClient(c), bucket: bucket}, nil
}

type presignReq struct {
    ContentType string `json:"contentType"`
    Size        int64  `json:"size"`
}

// POST /uploads/presign → {key, url}
func (u *Uploader) Presign(w http.ResponseWriter, r *http.Request) {
    var in presignReq
    if err := readJSON(r, &in); err != nil {
        writeErr(w, 400, "BAD_JSON", "Dữ liệu không đúng")
        return
    }
    ext, ok := allowed[in.ContentType]
    if !ok {
        writeErr(w, 400, "BAD_TYPE", "Chỉ nhận ảnh JPG, PNG, WebP hoặc PDF")
        return
    }
    if in.Size <= 0 || in.Size > maxSize {
        writeErr(w, 400, "TOO_LARGE", "Mỗi file tối đa 5MB")
        return
    }
    userID := userIDFrom(r.Context())
    // Server TỰ đặt tên khoá: người dùng không chọn được đường dẫn (không ghi đè được file của người khác)
    key := fmt.Sprintf("uploads/%d/%s%s", userID, ulid.Make(), ext)

    req, err := u.presig.PresignPutObject(r.Context(), &s3.PutObjectInput{
        Bucket:        aws.String(u.bucket),
        Key:           aws.String(key),
        ContentType:   aws.String(in.ContentType), // ký cả kiểu: upload kiểu khác thì S3 từ chối
        ContentLength: aws.Int64(in.Size),         // ký cả kích thước: gửi file to hơn thì bị từ chối
    }, s3.WithPresignExpires(5*time.Minute))
    if err != nil {
        writeErr(w, 500, "INTERNAL", "Lỗi máy chủ")
        return
    }
    writeJSON(w, 200, map[string]string{"key": key, "url": req.URL})
}
frontend
async function upload(file: File): Promise<string> {
  const { key, url } = await api.post("/uploads/presign", { contentType: file.type, size: file.size });

Khi client báo "đã nộp kèm khoá X", server phải kiểm tra lại:

Go · trích đoạn
func (s *Service) Submit(ctx context.Context, userID, assignmentID int64, keys []string) error {
    prefix := fmt.Sprintf("uploads/%d/", userID)
    for _, k := range keys {
        if !strings.HasPrefix(k, prefix) { // khoá phải do chính người này upload
            return ErrForbidden
        }
        head, err := s.s3.HeadObject(ctx, &s3.HeadObjectInput{Bucket: &s.bucket, Key: &k})
        if err != nil {
            return ErrFileMissing // client nói đã upload nhưng chưa có file thật
        }
        if aws.ToInt64(head.ContentLength) > maxSize {
            return ErrTooLarge
        }
    }
    return s.store.Submit(ctx, userID, assignmentID, keys)
}

Cho xem file: URL đọc có hạn

Kho file để riêng tư (không ai đọc được nếu không có chữ ký). Khi giáo viên mở bài nộp, server kiểm tra quyền rồi cấp URL đọc có hạn ngắn:

Go · trích đoạn
func (u *Uploader) ViewURL(ctx context.Context, key string) (string, error) {
    req, err := u.presig.PresignGetObject(ctx, &s3.GetObjectInput{
        Bucket:                     aws.String(u.bucket),
        Key:                        aws.String(key),
        ResponseContentDisposition: aws.String("inline"), // mở trong trình duyệt thay vì tải về
    }, s3.WithPresignExpires(10*time.Minute))
    if err != nil {
        return "", err
    }
    return req.URL, nil
}

Đừng tin Content-Type do trình duyệt gửi. Kẻ xấu có thể gửi file HTML với kiểu image/png. Bảo vệ: phục vụ file từ một tên miền khác với ứng dụng (ví dụ files.sobaitap-cdn.net), luôn trả header X-Content-Type-Options: nosniff, và với ảnh thì nên có bước xử lý lại (đọc bằng image.Decode, thu nhỏ, mã hoá lại) để vừa kiểm tra đúng là ảnh, vừa xoá thông tin vị trí GPS trong EXIF.

Dọn rác: người dùng xin URL, upload, rồi không bấm nộp. File nằm mãi trong kho. Đặt lifecycle rule xoá file trong uploads/ sau 1 ngày, và khi nộp thành công thì chép / đổi tên sang submissions/. Hoặc lưu bảng "file chờ" và có việc định kỳ dọn.

Khi buộc phải nhận file qua server

Ví dụ file Excel cần đọc ngay. Luôn giới hạn kích thước trước khi đọc:

Go · trích đoạn
func importExcel(w http.ResponseWriter, r *http.Request) {
    r.Body = http.MaxBytesReader(w, r.Body, 10<<20) // tối đa 10 MB, vượt thì đọc trả lỗi
    if err := r.ParseMultipartForm(2 << 20); err != nil { // 2 MB giữ trong RAM, phần còn lại ra file tạm
        writeErr(w, 413, "TOO_LARGE", "File tối đa 10MB")
        return
    }
    defer r.MultipartForm.RemoveAll() // xoá file tạm

    f, hdr, err := r.FormFile("file")
    if err != nil {
        writeErr(w, 400, "NO_FILE", "Chưa chọn file")
        return
    }
    defer f.Close()

    head := make([]byte, 512)
    n, _ := f.Read(head)
    kind := http.DetectContentType(head[:n]) // đoán kiểu từ nội dung, không tin tên file
    f.Seek(0, io.SeekStart)
    slog.Info("nhận file", "name", hdr.Filename, "size", hdr.Size, "kind", kind)
    // ... đọc Excel
}

#Gửi email

Không ai tự chạy máy chủ email để gửi cho khách hàng: email từ IP lạ gần như chắc chắn vào thư rác. Dùng dịch vụ: Amazon SES, Postmark, Resend, Mailgun, SendGrid. Bạn cần cấu hình SPF, DKIM, DMARC cho tên miền (các bản ghi DNS chứng minh "dịch vụ này được phép gửi thay tên miền của tôi").

Vì sao không gửi ngay trong handler: dịch vụ email có thể chậm 2 giây hoặc lỗi tạm thời, người dùng không nên chờ. Và nếu lỗi, cần thử lại mà không bắt người dùng bấm lại. Dùng hàng đợi của bài 23.

mail.go
package mail

import (
    "bytes"
    "context"
    "embed"
    "html/template"
    "net"
    "net/smtp"
)

//go:embed templates/*.html
var files embed.FS

var tpl = template.Must(template.ParseFS(files, "templates/*.html"))

type Message struct {
    To, Subject string
    HTML        string
}

// Render dùng html/template: tự escape tên người dùng, không chèn được mã HTML độc vào email
func Render(name string, data any) (string, error) {
    var buf bytes.Buffer
    if err := tpl.ExecuteTemplate(&buf, name, data); err != nil {
        return "", err
    }
    return buf.String(), nil
}

// Sender là interface: test dùng bản giả, dev dùng Mailpit, production dùng SES
type Sender interface {
    Send(ctx context.Context, m Message) error
}

type SMTPSender struct {
    Addr, User, Pass, From string
}

func (s SMTPSender) Send(ctx context.Context, m Message) error {
    body := "From: " + s.From + "\r\n" +
        "To: " + m.To + "\r\n" +
        "Subject: " + m.Subject + "\r\n" +
        "MIME-Version: 1.0\r\n" +
        "Content-Type: text/html; charset=UTF-8\r\n\r\n" + m.HTML
    host, _, _ := net.SplitHostPort(s.Addr)
    auth := smtp.PlainAuth("", s.User, s.Pass, host)
    return smtp.SendMail(s.Addr, auth, s.From, []string{m.To}, []byte(body))
}

Ví dụ trên dùng SMTP của thư viện chuẩn để bạn thấy bản chất (mọi dịch vụ đều nhận SMTP). Khi chạy thật, dùng SDK / API HTTP của dịch vụ (có id tin, có idempotency key, có timeout theo ctx). Tiêu đề có dấu tiếng Việt cần mã hoá: mime.QEncoding.Encode("utf-8", subject).

Bảng tin realtime bằng SSE cho nhiều người nghe

Viết server có GET /events?class=42 (SSE) và POST /submit?class=42&student=An. Mỗi lần có người nộp, mọi trình duyệt đang nghe lớp 42 nhận sự kiện submission. Lớp khác không nhận. Người nghe ngắt thì phải được gỡ khỏi danh sách.

Gợi ý từng bước

  1. Dùng lại cấu trúc Hub ở phần WebSocket nhưng đơn giản hơn: map[string]map[chan string]struct{} + mutex.
  2. Handler SSE: đăng ký một channel, defer gỡ ra, vòng select giữa channel và r.Context().Done().
  3. Mỗi tin: fmt.Fprintf(w, "event: submission\ndata: %s\n\n", msg) rồi Flush.
  4. Thử: mở 2 terminal curl -N lớp 42, 1 terminal lớp 43, terminal thứ tư gọi POST.
Xem lời giải
main.go
package main

import (
    "encoding/json"
    "fmt"
    "log"
    "net/http"
    "sync"
    "time"
)

type Hub struct {
    mu   sync.Mutex
    subs map[string]map[chan string]struct{}
}

func (h *Hub) add(room string) chan string {
    ch := make(chan string, 8)
    h.mu.Lock()
    defer h.mu.Unlock()
    if h.subs[room] == nil {
        h.subs[room] = map[chan string]struct{}{}
    }
    h.subs[room][ch] = struct{}{}
    return ch
}

func (h *Hub) remove(room string, ch chan string) {
    h.mu.Lock()
    defer h.mu.Unlock()
    delete(h.subs[room], ch)
    log.Printf("gỡ người nghe phòng %s, còn %d", room, len(h.subs[room]))
}

func (h *Hub) publish(room, msg string) int {
    h.mu.Lock()
    defer h.mu.Unlock()
    n := 0
    for ch := range h.subs[room] {
        select {
        case ch <- msg:
            n++
        default:
        }
    }
    return n
}

func main() {
    hub := &Hub{subs: map[string]map[chan string]struct{}{}}

    http.HandleFunc("GET /events", func(w http.ResponseWriter, r *http.Request) {
        room := r.URL.Query().Get("class")
        w.Header().Set("Content-Type", "text/event-stream")
        w.Header().Set("Cache-Control", "no-cache")
        rc := http.NewResponseController(w)

        ch := hub.add(room)
        defer hub.remove(room, ch)

        fmt.Fprint(w, ": connected\n\n") // dòng bắt đầu bằng ":" là chú thích, client bỏ qua
        rc.Flush()
        for {
            select {
            case msg := <-ch:
                fmt.Fprintf(w, "event: submission\ndata: %s\n\n", msg)
                if rc.Flush() != nil {
                    return
                }
            case <-r.Context().Done():
                return
            }
        }
    })

    http.HandleFunc("POST /submit", func(w http.ResponseWriter, r *http.Request) {
        q := r.URL.Query()
        b, _ := json.Marshal(map[string]string{"student": q.Get("student"), "at": time.Now().Format("15:04:05")})
        n := hub.publish(q.Get("class"), string(b))
        fmt.Fprintf(w, "đã báo cho %d người nghe\n", n)
    })

    log.Fatal(http.ListenAndServe(":8080", nil))
}

API cấp presigned URL chạy với MinIO trên máy

Chạy MinIO (S3 chạy trên máy) bằng Docker, viết POST /uploads/presign như trong bài, upload một ảnh bằng curl, rồi viết GET /files?key= trả URL đọc có hạn 10 phút.

Gợi ý từng bước

  1. MinIO: docker run -d -p 9000:9000 -p 9001:9001 -e MINIO_ROOT_USER=minio -e MINIO_ROOT_PASSWORD=minio12345 minio/minio server /data --console-address :9001. Tạo bucket sobaitap ở localhost:9001.
  2. Biến môi trường cho SDK: AWS_ACCESS_KEY_ID=minio, AWS_SECRET_ACCESS_KEY=minio12345, AWS_REGION=us-east-1, endpoint http://localhost:9000.
  3. Upload bằng curl: curl -X PUT -H "Content-Type: image/jpeg" --data-binary @anh.jpg "$URL". Kích thước file phải đúng bằng size đã xin.
  4. Thử gửi file lớn hơn hoặc đổi Content-Type: MinIO phải từ chối vì sai chữ ký.
Xem lời giải
main.go
package main

import (
    "context"
    "encoding/json"
    "fmt"
    "log"
    "net/http"
    "time"

    "github.com/aws/aws-sdk-go-v2/aws"
    "github.com/aws/aws-sdk-go-v2/config"
    "github.com/aws/aws-sdk-go-v2/service/s3"
)

const bucket = "sobaitap"

func main() {
    ctx := context.Background()
    cfg, err := config.LoadDefaultConfig(ctx)
    if err != nil {
        log.Fatal(err)
    }
    c := s3.NewFromConfig(cfg, func(o *s3.Options) {
        o.BaseEndpoint = aws.String("http://localhost:9000")
        o.UsePathStyle = true
    })
    ps := s3.NewPresignClient(c)
    n := 0

    http.HandleFunc("POST /uploads/presign", func(w http.ResponseWriter, r *http.Request) {
        var in struct {
            ContentType string `json:"contentType"`
            Size        int64  `json:"size"`
        }
        if json.NewDecoder(r.Body).Decode(&in) != nil || in.ContentType != "image/jpeg" || in.Size <= 0 || in.Size > 5<<20 {
            http.Error(w, "chỉ nhận ảnh JPG tối đa 5MB", 400)
            return
        }
        n++
        key := fmt.Sprintf("uploads/7/%d-%d.jpg", time.Now().Unix(), n) // bản thật dùng ULID như trong bài
        req, err := ps.PresignPutObject(r.Context(), &s3.PutObjectInput{
            Bucket: aws.String(bucket), Key: aws.String(key),
            ContentType: aws.String(in.ContentType), ContentLength: aws.Int64(in.Size),
        }, s3.WithPresignExpires(5*time.Minute))
        if err != nil {
            http.Error(w, err.Error(), 500)
            return
        }
        json.NewEncoder(w).Encode(map[string]string{"key": key, "url": req.URL})
    })

    http.HandleFunc("GET /files", func(w http.ResponseWriter, r *http.Request) {
        key := r.URL.Query().Get("key")
        // bản thật: kiểm tra người xem có quyền với file này
        req, err := ps.PresignGetObject(r.Context(), &s3.GetObjectInput{Bucket: aws.String(bucket), Key: aws.String(key)},
            s3.WithPresignExpires(10*time.Minute))
        if err != nil {
            http.Error(w, err.Error(), 500)
            return
        }
        json.NewEncoder(w).Encode(map[string]string{"url": req.URL})
    })

    log.Fatal(http.ListenAndServe(":8080", nil))
}
